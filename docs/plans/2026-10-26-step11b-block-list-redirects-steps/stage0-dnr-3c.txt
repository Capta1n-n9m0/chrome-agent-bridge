RESULT: (i) works from the worker itself (no offscreen document needed): existing file -> 200 (first chunk only, 1024 B
of a 5 MB file); missing file or missing folder -> rejects 'Failed to fetch' (5/5 each); a folder -> status 0 with a
listing, so the rule is "status 200 = a file". With file access off every path rejects (but #245 (b) refuses first).
(ii) is worse: a missing file attaches as size 0 and the page receives input+change BEFORE any check; a 1-byte slice
of it resolves (0 bytes), only a whole-file read rejects (NotFoundError), indistinguishable by size from an empty file
until read; {files: []} does NOT clear the input (files.length unchanged), value='' clears it without an event.
Chosen: (i).
--- scripts ---
// 0.11 3c (i): can an extension context tell an existing file from a missing one, reading no more than needed?
// toUrl mirrors what S0F would do: a Windows path -> file:///C:/seg/seg with each segment percent-encoded.
const ROOT = "%TEMP%\\claude\\s11b-stage0\\files\\";
const toUrl = (p) => "file:///" + p.split("\\").map((s, i) => (i === 0 ? s : encodeURIComponent(s))).join("/");
const probe = async (p) => {
  const t0 = performance.now();
  try {
    const r = await fetch(toUrl(p), { cache: "no-store" });
    const rd = r.body.getReader();
    const first = await rd.read();
    await rd.cancel();
    return { ok: r.ok, status: r.status, type: r.headers.get("content-type"), firstChunk: first.value ? first.value.length : 0, ms: Math.round(performance.now() - t0) };
  } catch (e) { return { rejected: String(e.message), ms: Math.round(performance.now() - t0) }; }
};
const red = (s) => s.replace(ROOT, "%TEMP%\\claude\\s11b-stage0\\files\\");
const paths = [ROOT + "made.txt", ROOT + "missing.txt", ROOT + "nodir\\missing.txt", ROOT + "sub dir", ROOT + "sub dir\\a b#\u00fc%.txt", ROOT + "big.bin"];
const out = { where: typeof document === "undefined" ? "service worker" : "offscreen document", fileAccess: chrome.extension?.isAllowedFileSchemeAccess ? await chrome.extension.isAllowedFileSchemeAccess() : "n/a", rows: [] };
for (const p of paths) {
  const res = [];
  for (let i = 0; i < 5; i++) res.push(await probe(p));
  const sig = res.map((r) => (r.rejected ? "rejected" : `${r.status}/${r.type}/${r.firstChunk}B`));
  out.rows.push({ path: red(p), url: red(toUrl(p).replace("file:///", "")), fiveRuns: sig.join(" | "), msMax: Math.max(...res.map((r) => r.ms)), err: res.find((r) => r.rejected)?.rejected });
}
return out;

  const pageState = async () => (await send("Runtime.evaluate", { expression: `JSON.stringify({ ev: __ev, n: document.getElementById('f').files.length, names: [...document.getElementById('f').files].map(x => x.name + ':' + x.size) })`, returnByValue: true })).result.value;
