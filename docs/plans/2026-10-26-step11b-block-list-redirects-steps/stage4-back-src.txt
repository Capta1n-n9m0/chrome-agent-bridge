=== ext-dnr/sw.js ===
// s11b back probe worker: builds a tab history the way the bridge does (tabs.create, then tabs.update with a wait for
// status "complete", handlers/navigate.ts + tabs.ts waitForLoad), then calls chrome.tabs.goBack and reads CDP
// Page.getNavigationHistory. Variants: active/background, the wait, a user activation, the navigation kind.
self.sleep = (ms) => new Promise((r) => setTimeout(r, ms));
self.waitLoad = (tabId, ms = 10000) => new Promise((res) => {
  let done = false;
  const fin = (why) => { if (!done) { done = true; chrome.tabs.onUpdated.removeListener(l); res(why); } };
  const l = (id, info) => { if (id === tabId && info.status === "complete") fin("complete"); };
  chrome.tabs.onUpdated.addListener(l);
  setTimeout(() => fin("timeout"), ms);
});
const cdp = async (tabId, fn) => {
  await chrome.debugger.attach({ tabId }, "1.3");
  try { return await fn((m, p = {}) => chrome.debugger.sendCommand({ tabId }, m, p)); }
  finally { await chrome.debugger.detach({ tabId }).catch(() => {}); }
};
const short = (u) => { try { const x = new URL(u); return `${x.port}${x.pathname}${x.search}${x.hash}`; } catch { return u; } };
self.hist = (tabId) => cdp(tabId, async (s) => {
  const h = await s("Page.getNavigationHistory");
  return { currentIndex: h.currentIndex, entries: h.entries.map((e) => `${short(e.url)} [${e.transitionType}]`) };
});
const gesture = (tabId) => cdp(tabId, (s) => s("Runtime.evaluate", { expression: "1", userGesture: true }));
const tryCall = async (f) => { try { await f(); return "ok"; } catch (e) { return `ERR ${e.message}`; } };

self.exp = async (o) => {
  const { active = false, mode = "wait", gest = false, nav = "update", urls, readBefore = false, fwdAfter = false } = o;
  const tab = await chrome.tabs.create({ url: urls[0], active });
  const id = tab.id;
  const first = await waitLoad(id);
  const steps = [first];
  for (const u of urls.slice(1)) {
    if (gest) await gesture(id);
    const p = waitLoad(id);
    if (nav === "update") await chrome.tabs.update(id, { url: u });
    else if (nav === "script") await chrome.scripting.executeScript({ target: { tabId: id }, func: (x) => { location.href = x; }, args: [u] });
    else if (nav === "link") await cdp(id, (s) => s("Runtime.evaluate", { expression: `(()=>{const a=document.getElementById('l');a.href=${JSON.stringify(u)};a.click();return 1})()`, userGesture: true }));
    else if (nav === "cdp") await cdp(id, (s) => s("Page.navigate", { url: u }));
    if (mode !== "nowait") steps.push(await p);
    if (mode === "settle") await sleep(1000);
  }
  if (mode === "nowait") await sleep(3000);
  const before = readBefore ? await hist(id) : null;
  const t0 = performance.now();
  const back = await tryCall(() => chrome.tabs.goBack(id));
  const backMs = Math.round(performance.now() - t0);
  await sleep(1500);
  const urlAfterBack = short((await chrome.tabs.get(id)).url);
  let fwd = null;
  if (fwdAfter && back === "ok") { fwd = await tryCall(() => chrome.tabs.goForward(id)); await sleep(1500); }
  const after = await hist(id);
  await chrome.tabs.remove(id);
  return { o: { active, mode, gest, nav, n: urls.length, readBefore }, loads: steps, back, backMs, urlAfterBack, fwd, before, after };
};
setInterval(() => chrome.runtime.getPlatformInfo(), 20000);
chrome.runtime.onMessage.addListener((m, s, reply) => { reply("awake"); });

=== tools/srv.py ===
"""s11b back probe fixture server: 127.0.0.1 only; every GET -> 200 small html with a full-width link #l.
Each request is appended to LOGFILE as: <METHOD> <path>.  usage: python -I srv.py PORT LOGFILE"""
import sys, threading
from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer

PORT = int(sys.argv[1]); LOG = sys.argv[2]; LOCK = threading.Lock()

class H(BaseHTTPRequestHandler):
    def do_GET(self):
        with LOCK:
            with open(LOG, "a", encoding="utf-8") as f:
                f.write(f"{self.command} {self.path}\n")
        body = b"<!doctype html><title>fx</title><a id=l href='#' style='display:block;height:400px'>link</a>"
        self.send_response(200); self.send_header("Content-Type", "text/html")
        self.send_header("Content-Length", str(len(body))); self.send_header("Cache-Control", "no-store"); self.end_headers()
        self.wfile.write(body)
    def log_message(self, *a): pass

class S(ThreadingHTTPServer):
    daemon_threads = True

S(("127.0.0.1", PORT), H).serve_forever()

// Launch a scratch Chrome on a temp profile (own --user-data-dir, #297) with a CDP pipe and port; load two probe
// extensions (ext-dnr carries declarativeNetRequestWithHostAccess, ext-plain does not); hold the pipe until
// <dir>/stop exists (or the limit), then Browser.close over the pipe. Adapted from stage0-dnr-ext-src.txt.
import { spawn } from "node:child_process";
import fs from "node:fs";
import path from "node:path";

const DIR = process.argv[2];
const PORT = process.argv[3];
const LIMIT_MS = Number(process.argv[4] || 1800000);
const CHROME = "C:\\Program Files\\Google\\Chrome\\Application\\chrome.exe";
const log = (s) => fs.appendFileSync(path.join(DIR, "logs", "launch.log"), `${new Date().toISOString()} ${s}\n`);

const args = [
  `--user-data-dir=${path.join(DIR, "profile")}`,
  "--remote-debugging-pipe",
  "--enable-unsafe-extension-debugging",
  `--remote-debugging-port=${PORT}`,
  "--no-first-run", "--no-default-browser-check", "--disable-background-networking", "--disable-component-update",
  "--disable-sync", "--disable-features=Translate,OptimizationHints,MediaRouter",
  "about:blank",
];
const cp = spawn(CHROME, args, { stdio: ["ignore", "ignore", "ignore", "pipe", "pipe"], windowsHide: false });
log(`spawned pid=${cp.pid}`);
fs.writeFileSync(path.join(DIR, "logs", "chrome.pid"), String(cp.pid));
const out = cp.stdio[3], inp = cp.stdio[4];
let buf = "", nextId = 1; const waiters = new Map();
inp.on("data", (d) => {
  buf += d.toString();
  let i;
  while ((i = buf.indexOf("\0")) >= 0) {
    const msg = JSON.parse(buf.slice(0, i)); buf = buf.slice(i + 1);
    if (msg.id && waiters.has(msg.id)) { waiters.get(msg.id)(msg); waiters.delete(msg.id); }
  }
});
const send = (method, params = {}) => new Promise((res) => { const id = nextId++; waiters.set(id, res); out.write(JSON.stringify({ id, method, params }) + "\0"); });
cp.on("exit", (code) => { log(`chrome exited code=${code}`); fs.writeFileSync(path.join(DIR, "logs", "closed"), String(code)); process.exit(0); });

const ver = await send("Browser.getVersion");
log(`version ${JSON.stringify(ver.result)}`);
const ids = {};
for (const name of ["ext-dnr", "ext-plain"]) {
  const r = await send("Extensions.loadUnpacked", { path: path.join(DIR, name) });
  log(`loadUnpacked ${name} ${JSON.stringify(r)}`);
  ids[name] = r.result?.id;
}
fs.writeFileSync(path.join(DIR, "logs", "ext.json"), JSON.stringify(ids));
const t0 = Date.now();
const iv = setInterval(async () => {
  if (fs.existsSync(path.join(DIR, "stop")) || Date.now() - t0 > LIMIT_MS) {
    clearInterval(iv);
    log("closing via Browser.close");
    const c = await send("Browser.close");
    log(`Browser.close -> ${JSON.stringify(c)}`);
  }
}, 1000);
