// S4a verifier-2 probes against the real module (copied into extension/test as zz-s4av2-probes.test.ts, deleted after).
// Each test asserts the SAFE / expected outcome; a red test is a finding.
import { describe, it, expect, vi } from "vitest";
import { ErrorCodes } from "@bridge/shared";
import { DebuggerKeeper, KEEPER_DETACH_WAIT_MS, KEEPER_IDLE_MS, type KeeperDeps } from "../src/debugger-keeper.js";
import { DebuggerSessions } from "../src/safety-debugger.js";
import { DISARMED_ALL } from "../src/safety-state.js";
import { describeDebuggerError } from "../src/debugger-errors.js";
import { type SafetyRefusal } from "../src/safety.js";
import { BridgeError } from "../src/errors.js";

const BLOCKED: SafetyRefusal = { code: ErrorCodes.DOMAIN_BLOCKED, message: "blocked (probe)" };
const mapError = (err: unknown): Error => {
  const d = describeDebuggerError(err, "Probe");
  return new BridgeError(d.code, d.message, d.data);
};
const flush = async () => {
  for (let i = 0; i < 40; i++) await Promise.resolve();
};

function clock() {
  let now = 0;
  let next = 1;
  const timers = new Map<number, { at: number; fn: () => void }>();
  return {
    timers: {
      set: (fn: () => void, ms: number) => {
        const id = next++;
        timers.set(id, { at: now + ms, fn });
        return id;
      },
      clear: (h: unknown) => void timers.delete(h as number),
    },
    async advance(ms: number) {
      const end = now + ms;
      for (;;) {
        const due = [...timers.entries()].filter(([, t]) => t.at <= end).sort((a, b) => a[1].at - b[1].at)[0];
        if (!due) break;
        timers.delete(due[0]);
        now = due[1].at;
        due[1].fn();
        await flush();
      }
      now = end;
      await flush();
    },
    pending: () => timers.size,
  };
}

/** Chrome: `ours` = this extension's session per tab; `devtools` = a DevTools client. getTargets reads either. */
function setup() {
  const ours = new Set<number>();
  const devtools = new Set<number>();
  let pendingAttach: (() => void) | null = null;
  let deferAttach = false;
  const c = clock();
  const sessions = new DebuggerSessions();
  let check: (id: number) => Promise<SafetyRefusal | null> = async () => null;
  let targets: () => Promise<{ tabId: number; attached: boolean }[]> = async () =>
    [...new Set([...ours, ...devtools])].map((tabId) => ({ tabId, attached: true }));
  const deps: KeeperDeps = {
    attach: vi.fn((id: number) => {
      if (ours.has(id) || devtools.has(id)) return Promise.reject(new Error(`Another debugger is already attached to the tab with id: ${id}.`));
      if (!deferAttach) {
        ours.add(id);
        return Promise.resolve();
      }
      return new Promise<void>((r) => (pendingAttach = () => (ours.add(id), r())));
    }),
    detach: vi.fn(async (id: number) => {
      if (!ours.delete(id)) throw new Error(`Debugger is not attached to the tab with id: ${id}.`);
    }),
    getTargets: vi.fn(() => targets()),
    timers: c.timers,
    sessions,
    tabCheck: (id) => check(id),
  };
  const keeper = new DebuggerKeeper(deps);
  /** a use whose work issues one CDP command (fails if our session is not attached) */
  const cmd = (id: number, v: unknown = "ok") =>
    keeper.use(id, async () => {
      if (!ours.has(id)) throw new Error(`Debugger is not attached to the tab with id: ${id}.`);
      return v;
    }, { mapError });
  return {
    ours, devtools, sessions, deps, keeper, clock: c, cmd,
    defer: () => (deferAttach = true),
    finishAttach: () => pendingAttach!(),
    setCheck: (f: typeof check) => (check = f),
    setTargets: (f: typeof targets) => (targets = f),
  };
}

describe("S4a verifier-2: round-1 probes (P4 with Chrome-faithful getTargets)", () => {
  it("P1 disarm while the attach is pending", async () => {
    const t = setup();
    t.defer();
    const p = t.cmd(7);
    p.catch(() => {});
    await flush();
    await t.sessions.detachAll(DISARMED_ALL, t.deps.detach);
    t.finishAttach();
    await expect(p).rejects.toMatchObject({ code: ErrorCodes.DISARMED });
    await flush();
    expect(t.ours.has(7)).toBe(false);
    expect(t.sessions.size).toBe(0);
    expect(t.clock.pending()).toBe(0);
  });

  it("P2 refusing recheck while the attach is pending", async () => {
    const t = setup();
    t.defer();
    const p = t.cmd(7);
    p.catch(() => {});
    await flush();
    t.setCheck(async () => BLOCKED);
    await t.keeper.recheckTab(7);
    t.finishAttach();
    await expect(p).rejects.toMatchObject({ code: ErrorCodes.DOMAIN_BLOCKED });
    await flush();
    expect(t.ours.has(7)).toBe(false);
    expect(t.sessions.size).toBe(0);
    expect(t.clock.pending()).toBe(0);
  });

  it("P3 another call's failing guard on a busy tab: busy use keeps its answer", async () => {
    const t = setup();
    let release!: () => void;
    const gate = new Promise<void>((r) => (release = r));
    const a = t.keeper.use(7, async () => {
      await gate;
      if (!t.ours.has(7)) throw new Error("Detached while handling command.");
      return "a";
    }, { mapError });
    a.catch(() => {});
    await flush();
    const b = t.keeper.use(7, async () => "b", { guard: { effect: async () => { throw new Error("unexpected guard failure"); } }, mapError });
    await expect(b).rejects.toThrow();
    release();
    expect(await a).toBe("a");
    await flush();
    expect(t.ours.has(7)).toBe(false);
    expect(t.sessions.size).toBe(0);
  });

  it("P4 late onDetach must not orphan a newer session (getTargets faithful)", async () => {
    const t = setup();
    await t.cmd(7, 1);
    t.ours.delete(7); // Chrome ended session 1; onDetach queued
    await t.cmd(7).catch(() => {}); // fails, drops
    await t.cmd(7, 3); // session 2
    expect(t.ours.has(7)).toBe(true);
    t.keeper.onDetach(7, "canceled_by_user");
    await flush();
    await t.sessions.detachAll(DISARMED_ALL, t.deps.detach);
    expect(t.ours.has(7)).toBe(false);
  });

  it("P4' the round-1 probe's own fake (getTargets always []) — expected red, by construction", async () => {
    const t = setup();
    t.setTargets(async () => []);
    await t.cmd(7, 1);
    t.ours.delete(7);
    await t.cmd(7).catch(() => {});
    await t.cmd(7, 3);
    t.keeper.onDetach(7, "canceled_by_user");
    await flush();
    await t.sessions.detachAll(DISARMED_ALL, t.deps.detach);
    expect(t.ours.has(7)).toBe(false);
  });

  it("P5 page blocked between guard and entry", async () => {
    const t = setup();
    const guard = { effect: async () => { t.setCheck(async () => BLOCKED); await t.keeper.recheckTab(7); } };
    await t.keeper.use(7, async () => "x", { guard, mapError });
    expect(t.keeper.state(7) === "idle" && t.ours.has(7)).toBe(false);
    expect(t.sessions.size).toBe(0);
    expect(t.clock.pending()).toBe(0);
  });
});

describe("S4a verifier-2: attacks on the fixes", () => {
  it("A1 killed busy use's second detach must not detach a NEWER session (sequential C must not fail)", async () => {
    const t = setup();
    let release!: () => void;
    const gate = new Promise<void>((r) => (release = r));
    // A: a long use in a non-CDP wait (e.g. trusted input's pause, upload's file check)
    const a = t.keeper.use(7, async () => { await gate; return "a"; }, { mapError });
    a.catch(() => {});
    await flush();
    // a transient tab-check failure (UNREADABLE) or a blocked hop kills A
    t.setCheck(async () => { throw new Error("tabs.get failed"); });
    await t.keeper.recheckTab(7);
    t.setCheck(async () => null);
    // B: a new call on the tab whose guard passes; A is unmapped, so B attaches fresh
    expect(await t.cmd(7, "b")).toBe("b");
    expect(t.keeper.state(7)).toBe("idle");
    expect(t.ours.has(7)).toBe(true);
    // A ends
    release();
    await expect(a).rejects.toMatchObject({ code: ErrorCodes.DISARMED });
    await flush();
    // B's kept session must still be attached, or C (a sequential call) answers a false error
    expect(t.ours.has(7)).toBe(t.keeper.state(7) !== "none");
    expect(await t.cmd(7, "c")).toBe("c");
  });

  it("A2 onDetach's getTargets slower than the 1 s wait bound must not detach the next session", async () => {
    const t = setup();
    await t.cmd(7);
    t.ours.delete(7); // Chrome really ended it
    let answer!: () => void;
    t.setTargets(() => new Promise((r) => (answer = () => r([...t.ours].map((tabId) => ({ tabId, attached: true }))))));
    t.keeper.onDetach(7, "canceled_by_user");
    const b = t.cmd(7, "b");
    await flush();
    await t.clock.advance(KEEPER_DETACH_WAIT_MS);
    expect(await b).toBe("b"); // B attached after the bound
    answer(); // getTargets now reads B's session as attached
    await flush();
    expect(t.ours.has(7)).toBe(t.keeper.state(7) !== "none");
    expect(await t.cmd(7, "c")).toBe("c");
  });

  it("A3 the post-attach check while busy: an overlapping call answers DEBUGGER_DEVTOOLS, 1 attach; then idle, one timer", async () => {
    const t = setup();
    let answer!: (r: SafetyRefusal | null) => void;
    t.setCheck(() => new Promise((r) => (answer = r)));
    const a = t.cmd(7, "a");
    await flush();
    expect(t.keeper.state(7)).toBe("busy");
    await expect(t.cmd(7, "b")).rejects.toMatchObject({ code: ErrorCodes.DEBUGGER_DEVTOOLS });
    expect(t.deps.attach).toHaveBeenCalledTimes(1);
    answer(null);
    expect(await a).toBe("a");
    expect(t.keeper.state(7)).toBe("idle");
    expect(t.clock.pending()).toBe(1);
    await t.clock.advance(KEEPER_IDLE_MS);
    expect(t.ours.has(7)).toBe(false);
    expect(t.sessions.size).toBe(0);
    expect(t.clock.pending()).toBe(0);
  });

  it("A4 a tabCheck that never settles holds the use's reply (no bound in the keeper) — documents the dependency", async () => {
    const t = setup();
    t.setCheck(() => new Promise(() => {}));
    let done = false;
    void t.cmd(7).then(() => (done = true));
    await flush();
    await t.clock.advance(60_000);
    expect(done).toBe(false); // expected: the keeper relies on S4b's tabCheck being bounded
  });

  it("A5 F3 with DevTools taking over: the extra detach leaves DevTools attached; next use answers DEBUGGER_DEVTOOLS; balance", async () => {
    const t = setup();
    await t.cmd(7);
    t.ours.delete(7);
    t.devtools.add(7); // replaced_with_devtools
    t.keeper.onDetach(7, "replaced_with_devtools");
    await flush();
    expect(t.deps.detach).toHaveBeenCalledTimes(1); // the no-op detach
    expect(t.devtools.has(7)).toBe(true);
    await expect(t.cmd(7)).rejects.toMatchObject({ code: ErrorCodes.DEBUGGER_DEVTOOLS });
    expect(t.sessions.size).toBe(0);
    expect(t.clock.pending()).toBe(0);
    expect(t.keeper.state(7)).toBe("none");
  });

  it("A6 release paths: tabGone, recheck, drop, idle, disarm on idle — registry and timers balanced, no orphan", async () => {
    const t = setup();
    await t.cmd(1); await t.keeper.tabGone(1);
    await t.cmd(2); t.setCheck(async () => BLOCKED); await t.keeper.recheckTab(2); t.setCheck(async () => null);
    await t.keeper.use(3, async (u) => { u.drop(); return 1; }, { mapError });
    await t.cmd(4); await t.clock.advance(KEEPER_IDLE_MS);
    await t.cmd(5); await t.sessions.detachAll(DISARMED_ALL, t.deps.detach); t.sessions.refuse(null);
    await flush();
    expect(t.keeper.size).toBe(0);
