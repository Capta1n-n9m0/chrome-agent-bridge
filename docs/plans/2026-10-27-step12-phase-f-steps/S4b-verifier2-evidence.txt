// S4b verifier 2 probes (scratch only, never committed)
import { describe, it, expect } from "vitest";
import { ErrorCodes } from "@bridge/shared";
import { DebuggerKeeper } from "../src/debugger-keeper.js";
import { DebuggerSessions } from "../src/safety-debugger.js";
import { refusalError, type SafetyRefusal } from "../src/safety.js";

const BLOCKED: SafetyRefusal = { code: ErrorCodes.DOMAIN_BLOCKED, message: "blocked (test)" };
const flush = async () => {
  for (let i = 0; i < 30; i++) await Promise.resolve();
};
const realTimers = { set: (fn: () => void, ms: number) => setTimeout(fn, ms), clear: (h: unknown) => clearTimeout(h as ReturnType<typeof setTimeout>) };

/** A fake Chrome with one debugger per tab: attach throws while still attached; detach lands when released. */
function chromeLike(order: string[]) {
  let attached = false;
  const releases: (() => void)[] = [];
  return {
    releases,
    deps: {
      attach: async () => {
        if (attached) {
          order.push("attach:ANOTHER");
          throw new Error("Another debugger is already attached to the tab with id: 7.");
        }
        attached = true;
        order.push("attach");
      },
      detach: () =>
        new Promise<void>((r) =>
          releases.push(() => {
            attached = false;
            order.push("detached");
            r();
          }),
        ),
      getTargets: async () => [],
      timers: realTimers,
      sessions: new DebuggerSessions(),
      tabCheck: async () => null,
    },
  };
}

describe("S4b verifier 2 probes", () => {
  it("P-LOOP: a detach that starts during the guard's own await is waited for, and the guard runs again", async () => {
    const order: string[] = [];
    const c = chromeLike(order);
    const keeper = new DebuggerKeeper(c.deps);
    let slow: (() => void) | null = null;
    let guards = 0;
    const guard = {
      effect: async () => {
        guards++;
        order.push("guard");
        if (guards === 2) await new Promise<void>((r) => (slow = r));
      },
    };
    const opts = { guard, mapError: (e: unknown) => e as Error };
    await keeper.use(7, async () => void order.push("fx1"), opts); // kept idle
    const second = keeper.use(7, async () => void order.push("fx2"), opts);
    second.catch(() => {});
    await flush();
    expect(slow).not.toBeNull(); // the second guard is awaiting
    void keeper.tabGone(7); // a detach starts during the guard
    await flush();
    slow!();
    await flush();
    c.releases.forEach((r) => r());
    let out: unknown = "ok";
    try {
      await second;
    } catch (e) {
      out = (e as Error).message;
    }
    console.log("P-LOOP", JSON.stringify({ out, order, guards }));
    expect(out).toBe("ok");
    expect(order).not.toContain("attach:ANOTHER");
  });

  it("P-LOOP-REFUSE: a refusal that lands during the loop's second wait refuses the use", async () => {
    const order: string[] = [];
    const c = chromeLike(order);
    const keeper = new DebuggerKeeper(c.deps);
    let slow: (() => void) | null = null;
    let guards = 0;
    let blocked = false;
    const guard = {
      effect: async () => {
        guards++;
        if (blocked) throw refusalError(BLOCKED);
        if (guards === 2) await new Promise<void>((r) => (slow = r));
      },
    };
    const opts = { guard, mapError: (e: unknown) => e as Error };
    await keeper.use(7, async () => void order.push("fx1"), opts);
    const second = keeper.use(7, async () => void order.push("fx2"), opts);
    second.catch(() => {});
    await flush();
    void keeper.tabGone(7);
    await flush();
    slow!();
    await flush();
    blocked = true; // during the wait for the newer detach
    c.releases.forEach((r) => r());
    await expect(second).rejects.toMatchObject({ code: ErrorCodes.DOMAIN_BLOCKED });
    console.log("P-LOOP-REFUSE", JSON.stringify({ order, guards }));
    expect(order).toEqual(["attach", "fx1", "detached"]);
  });

  it("P-BOUND: a detach that never settles holds a use 1 s once, not once per loop turn (F5)", async () => {
    const order: string[] = [];
    const c = chromeLike(order);
    const keeper = new DebuggerKeeper(c.deps);
    let guards = 0;
    const guard = { effect: async () => void guards++ };
    const opts = { guard, mapError: (e: unknown) => e as Error };
    await keeper.use(7, async () => void order.push("fx1"), opts);
    void keeper.tabGone(7); // never released
    const t0 = Date.now();
    let out: unknown = "ok";
    try {
      await keeper.use(7, async () => void order.push("fx2"), opts);
    } catch (e) {
      out = (e as Error).message;
    }
    const ms = Date.now() - t0;
    console.log("P-BOUND", JSON.stringify({ ms, out, order, guards }));
    expect(ms).toBeLessThan(1500);
    expect(guards).toBe(2);
  });
});
