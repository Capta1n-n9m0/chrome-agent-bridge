// SCRATCH (S4a verifier 3): probes; deleted after the run.
import { describe, it, expect, vi } from "vitest";
import { ErrorCodes } from "@bridge/shared";
import { DebuggerKeeper, KEEPER_DETACH_WAIT_MS, type KeeperDeps } from "../src/debugger-keeper.js";
import { DebuggerSessions } from "../src/safety-debugger.js";
import { DISARMED_ALL } from "../src/safety-state.js";
import { describeDebuggerError } from "../src/debugger-errors.js";
import { refusalError, type SafetyRefusal } from "../src/safety.js";
import { BridgeError } from "../src/errors.js";

const BLOCKED: SafetyRefusal = { code: ErrorCodes.DOMAIN_BLOCKED, message: "blocked (test)" };
const mapError = (err: unknown): Error => {
  const d = describeDebuggerError(err, "Test");
  return new BridgeError(d.code, d.message, d.data);
};
async function flush(): Promise<void> {
  for (let i = 0; i < 30; i++) await Promise.resolve();
}
function fakeClock() {
  let now = 0;
  const timers = new Map<number, { at: number; fn: () => void }>();
  let next = 1;
  return {
    timers: {
      set: (fn: () => void, ms: number) => {
        const id = next++;
        timers.set(id, { at: now + ms, fn });
        return id;
      },
      clear: (h: unknown) => void timers.delete(h as number),
    },
    async advance(ms: number) {
      const end = now + ms;
      for (;;) {
        const due = [...timers.entries()].filter(([, t]) => t.at <= end).sort((a, b) => a[1].at - b[1].at)[0];
        if (!due) break;
        timers.delete(due[0]);
        now = due[1].at;
        due[1].fn();
        await flush();
      }
      now = end;
      await flush();
    },
    pending: () => timers.size,
  };
}

/** Chrome with a queue of pending operations; `step(kind?)` completes the first (of that kind). */
function queuedChrome() {
  const attached = new Set<number>();
  type Op = { kind: "attach" | "detach" | "targets"; tabId?: number; res: (v?: unknown) => void; rej: (e: unknown) => void };
  const q: Op[] = [];
  const op = (kind: Op["kind"], tabId?: number) => new Promise<any>((res, rej) => q.push({ kind, tabId, res, rej }));
  const calls = { attach: 0, detach: 0, targets: 0 };
  return {
    attached,
    q,
    calls,
    deps: {
      attach: (id: number) => (calls.attach++, op("attach", id)),
      detach: (id: number) => (calls.detach++, op("detach", id)),
      getTargets: () => (calls.targets++, op("targets")),
    },
    async step(kind?: Op["kind"]) {
      const i = kind ? q.findIndex((o) => o.kind === kind) : 0;
      if (i < 0 || !q[i]) throw new Error(`no pending ${kind ?? "op"}`);
      const [o] = q.splice(i, 1);
      if (o.kind === "attach") {
        if (attached.has(o.tabId!)) o.rej(new Error(`Another debugger is already attached to the tab with id: ${o.tabId}.`));
        else (attached.add(o.tabId!), o.res());
      } else if (o.kind === "detach") {
        if (attached.delete(o.tabId!)) o.res();
        else o.rej(new Error(`Debugger is not attached to the tab with id: ${o.tabId}.`));
      } else o.res([...attached].map((tabId) => ({ tabId, attached: true })));
      await flush();
    },
    async drain() {
      while (q.length) await this.step();
    },
  };
}

function setupQ() {
  const c = queuedChrome();
  const clock = fakeClock();
  const sessions = new DebuggerSessions();
  let check: (tabId: number) => Promise<SafetyRefusal | null> = async () => null;
  const deps: KeeperDeps = { ...c.deps, timers: clock.timers, sessions, tabCheck: (id) => check(id) };
  const keeper = new DebuggerKeeper(deps);
  const guard = { effect: vi.fn(async () => {}) };
  const cdp = (tabId: number, v: unknown) =>
    keeper.use(tabId, async () => {
      if (!c.attached.has(tabId)) throw new Error(`Debugger is not attached to the tab with id: ${tabId}.`);
      return v;
    }, { guard, mapError });
  return { c, clock, sessions, deps, keeper, guard, cdp, setCheck: (f: typeof check) => (check = f) };
}

/** A use that settles; drains Chrome while waiting. */
async function run<T>(t: ReturnType<typeof setupQ>, p: Promise<T>): Promise<T> {
  let done = false;
  p.then(() => (done = true), () => (done = true));
  for (let i = 0; i < 50 && !done; i++) {
    await flush();
    if (t.c.q.length) await t.c.step();
  }
  return p;
}

/** No Chrome session without a registry record; after a disarm nothing attached. */
async function reach(t: ReturnType<typeof setupQ>) {
  await t.c.drain();
  await flush();
  const orphan = t.c.attached.has(7) && t.sessions.size === 0;
  const p = t.sessions.detachAll(DISARMED_ALL, t.deps.detach);
  await t.c.drain();
  await p;
  return { orphan, afterDisarm: t.c.attached.has(7) };
}

describe("V3 N2 pending case", () => {
  it("Q1 late onDetach while the next use's attach is pending: the orphan is detached (real) — alt skips it", async () => {
    const t = setupQ();
    expect(await run(t, t.cdp(7, "x"))).toBe("x"); // session X kept and attached
    t.keeper.onDetach(7, "canceled_by_user"); // late event for an older session: X is forgotten but still attached
    expect(t.sessions.size).toBe(0);
    const b = t.cdp(7, "b");
    b.catch(() => {});
    await flush();
    await t.clock.advance(KEEPER_DETACH_WAIT_MS); // B stops waiting; its attach is queued after getTargets
    expect(t.c.q.map((o) => o.kind)).toEqual(["targets", "attach"]);
    expect(t.keeper.state(7)).toBe("busy");
    await t.c.step(); // getTargets answers while B's attach is pending
    await t.c.drain();
    await expect(b).rejects.toMatchObject({ code: ErrorCodes.DEBUGGER_DEVTOOLS }); // today's overlap answer
    const r = await reach(t);
    expect(r).toEqual({ orphan: false, afterDisarm: false });
  });

  it("Q1b same, B's attach landing in Chrome before getTargets answers but resolving after: B's session detached under it", async () => {
    // a 'pending' entry whose attach already landed in Chrome: the check reads it attached and detaches it
    const t = setupQ();
    expect(await run(t, t.cdp(7, "x"))).toBe("x");
    t.c.attached.delete(7); // the event is real: Chrome ended X
    t.keeper.onDetach(7, "canceled_by_user");
    const b = t.keeper.use(7, async () => "b", { guard: t.guard, mapError });
    b.catch(() => {});
    await flush();
    await t.clock.advance(KEEPER_DETACH_WAIT_MS);
    // Chrome: B's attach completes, then getTargets is answered, then B's attach reply reaches the worker
    const att = t.c.q.find((o) => o.kind === "attach")!;
    t.c.q.splice(t.c.q.indexOf(att), 1);
    t.c.attached.add(7);
    await t.c.step("targets");
    await flush();
    att.res();
    await t.c.drain();
    const ans = await b.then((v) => v, (e) => e.code);
    const next = await run(t, t.cdp(7, "c")).then((v) => v, (e) => e.code);
    const r = await reach(t);
    console.log("V3", JSON.stringify({ ans, next, ...r }));
    expect({ ans, next, ...r }).toEqual({ ans: "b", next: "c", orphan: false, afterDisarm: false });
  });
});

describe("V3 onDetach on a pending entry (F3 class)", () => {

    const t = setupQ();

    t.c.attached.delete(7); // Chrome ended X (Cancel); its onDetach is still queued
    await run(t, t.cdp(7, "fails")).catch(() => {}); // X's use fails, X released
    expect(t.keeper.state(7)).toBe("none");
    const a = t.keeper.use(7, async () => "a", { guard: t.guard, mapError });

    await t.c.step("targets"); // Chrome answers getTargets at once: A's attach has not landed
    await t.c.step("attach"); // A's attach lands
    const ans = await a.then((v) => v, (e) => e.code);
    const r = await reach(t);
    console.log("V3", JSON.stringify({ ans, ...r, size: t.sessions.size }));
    expect({ ans, ...r }).toEqual({ ans: "a", orphan: false, afterDisarm: false });
  });

  it("Q2b same, Chrome answers in order (attach before getTargets)", async () => {

    t.c.attached.delete(7);

    console.log("V3", JSON.stringify({ ans, next, ...r }));
    expect({ ans, next, ...r }).toEqual({ ans: "a", next: "c", orphan: false, afterDisarm: false });
  });
});

describe("V3 kill matrix: no killed session stays attached outside the registry", () => {
  type Kill = "disarm" | "recheckBlocked" | "recheckThrow" | "guardBlocked" | "guardDisarmed";
  type When = "pendingDetachFirst" | "pendingAttachFirst" | "busy" | "idle" | "duringP5";
  const kills: Kill[] = ["disarm", "recheckBlocked", "recheckThrow", "guardBlocked", "guardDisarmed"];
  const whens: When[] = ["pendingDetachFirst", "pendingAttachFirst", "busy", "idle", "duringP5"];
  async function kill(t: ReturnType<typeof setupQ>, k: Kill) {
    if (k === "disarm") return void t.sessions.detachAll(DISARMED_ALL, t.deps.detach);
    if (k.startsWith("recheck")) {
      t.setCheck(k === "recheckBlocked" ? async () => BLOCKED : async () => Promise.reject(new Error("x")));
      return void t.keeper.recheckTab(7);
    }
    const r = k === "guardBlocked" ? BLOCKED : DISARMED_ALL;
    void t.keeper.use(7, async () => "z", { guard: { effect: async () => Promise.reject(refusalError(r)) }, mapError }).catch(() => {});
  }
  for (const k of kills)
    for (const w of whens)
      it(`${k} × ${w}`, async () => {
        const t = setupQ();
        let a: Promise<unknown>;
        let gate!: () => void;
        if (w === "idle") {
          expect(await run(t, t.cdp(7, "a"))).toBe("a");
          a = Promise.resolve();
          await kill(t, k);
        } else if (w === "duringP5") {
          let releaseCheck!: (v: SafetyRefusal | null) => void;
          t.setCheck(() => new Promise((r) => (releaseCheck = r)));
          a = t.cdp(7, "a");
          a.catch(() => {});
          await flush();
          await t.c.step("attach");
          await flush(); // A in its finally, awaiting the P5 check
          await kill(t, k);
          await flush();
          releaseCheck(null);
        } else {
          const g = new Promise<void>((r) => (gate = r));
          a = t.keeper.use(7, async () => (await g, "a"), { guard: t.guard, mapError });
          a.catch(() => {});
          await flush();
          if (w === "busy") await t.c.step("attach");
          await kill(t, k);
          await flush();
