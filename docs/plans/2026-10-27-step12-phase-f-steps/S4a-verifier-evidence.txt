// S4a verifier probes against the real module (copied into extension/test as zz-s4av-probes.test.ts, deleted after).
// Each test asserts the SAFE outcome; a red test is a finding.
import { describe, it, expect, vi } from "vitest";
import { ErrorCodes } from "@bridge/shared";
import { DebuggerKeeper, type KeeperDeps } from "../src/debugger-keeper.js";
import { DebuggerSessions } from "../src/safety-debugger.js";
import { DISARMED_ALL } from "../src/safety-state.js";
import { describeDebuggerError } from "../src/debugger-errors.js";
import { type SafetyRefusal } from "../src/safety.js";
import { BridgeError } from "../src/errors.js";

const BLOCKED: SafetyRefusal = { code: ErrorCodes.DOMAIN_BLOCKED, message: "blocked (probe)" };
const mapError = (err: unknown): Error => {
  const d = describeDebuggerError(err, "Probe");
  return new BridgeError(d.code, d.message, d.data);
};
const flush = async () => {
  for (let i = 0; i < 30; i++) await Promise.resolve();
};

/** attachMode "deferred": Chrome's attached state is set only when the attach promise resolves. */
function setup() {
  const attached = new Set<number>();
  let pendingAttach: (() => void) | null = null;
  let deferAttach = false;
  const sessions = new DebuggerSessions();
  let check: (id: number) => Promise<SafetyRefusal | null> = async () => null;
  const timers = { set: (_fn: () => void, _ms: number) => ({}), clear: (_h: unknown) => {} };
  const deps: KeeperDeps = {
    attach: vi.fn((id: number) => {
      if (attached.has(id)) return Promise.reject(new Error(`Another debugger is already attached to the tab with id: ${id}.`));
      if (!deferAttach) {
        attached.add(id);
        return Promise.resolve();
      }
      return new Promise<void>((r) => (pendingAttach = () => (attached.add(id), r())));
    }),
    detach: vi.fn(async (id: number) => {
      if (!attached.delete(id)) throw new Error(`Debugger is not attached to the tab with id: ${id}.`);
    }),
    getTargets: vi.fn(async () => []),
    timers,
    sessions,
    tabCheck: (id) => check(id),
  };
  const keeper = new DebuggerKeeper(deps);
  return {
    attached, sessions, deps, keeper,
    defer: () => (deferAttach = true),
    finishAttach: () => pendingAttach!(),
    setCheck: (f: typeof check) => (check = f),
  };
}

describe("S4a verifier probes", () => {
  it("P1 disarm while the attach is pending: no session stays attached after the disarm", async () => {
    const t = setup();
    t.defer();
    const p = t.keeper.use(7, async () => "x", { mapError });
    p.catch(() => {});
    await flush();
    await t.sessions.detachAll(DISARMED_ALL, t.deps.detach); // Chrome: "not attached" yet (ignored)
    t.finishAttach();
    await expect(p).rejects.toMatchObject({ code: ErrorCodes.DISARMED });
    await flush();
    expect(t.attached.has(7)).toBe(false);
  });

  it("P2 a refusing recheck while the attach is pending: no session stays attached on the blocked tab", async () => {
    const t = setup();
    t.defer();
    const p = t.keeper.use(7, async () => "x", { mapError });
    p.catch(() => {});
    await flush();
    t.setCheck(async () => BLOCKED);
    await t.keeper.recheckTab(7);
    t.finishAttach();
    await expect(p).rejects.toMatchObject({ code: ErrorCodes.DOMAIN_BLOCKED });
    await flush();
    expect(t.attached.has(7)).toBe(false);
  });

  it("P3 another call's failing guard on a busy tab: the busy use keeps its own answer (or answers the refusal)", async () => {
    const t = setup();
    let release!: () => void;
    const gate = new Promise<void>((r) => (release = r));
    const a = t.keeper.use(7, async () => {
      await gate;
      if (!t.attached.has(7)) throw new Error("Detached while handling command.");
      return "a";
    }, { mapError });
    a.catch(() => {});
    await flush();
    const b = t.keeper.use(7, async () => "b", { guard: { effect: async () => { throw new Error("unexpected guard failure"); } }, mapError });
    await expect(b).rejects.toThrow();
    release();
    const outA = await a.then((v) => v, (e) => (e as { code?: string }).code);
    expect(outA).toBe("a");
  });

  it("P4 a late onDetach for an old session must not orphan a newer one outside the registry", async () => {
    const t = setup();
    await t.keeper.use(7, async () => 1, { mapError }); // session 1 kept
    t.attached.delete(7); // Chrome ended session 1 (Cancel); its onDetach is still queued
    await t.keeper.use(7, async () => { throw new Error("Debugger is not attached to the tab with id: 7."); }, { mapError }).catch(() => {});
    await t.keeper.use(7, async () => 3, { mapError }); // session 2 attached, kept
    expect(t.attached.has(7)).toBe(true);
    t.keeper.onDetach(7, "canceled_by_user"); // the late event for session 1
    await t.sessions.detachAll(DISARMED_ALL, t.deps.detach);
    expect(t.attached.has(7)).toBe(false); // the disarm must reach every attached session
  });

  it("P5 the page becomes blocked between the guard's read and the attach: no idle session is left on the blocked page", async () => {
    const t = setup();
    const guard = {
      effect: async () => {
        // the guard has read the tab as allowed; now tabs.onUpdated lands with a blocked URL
        t.setCheck(async () => BLOCKED);
        await t.keeper.recheckTab(7);
      },
    };
    await t.keeper.use(7, async () => "x", { guard, mapError });
    expect(t.keeper.state(7) === "idle" && t.attached.has(7)).toBe(false);
  });
});
