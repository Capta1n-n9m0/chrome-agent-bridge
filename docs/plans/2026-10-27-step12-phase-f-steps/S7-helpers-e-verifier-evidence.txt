    res.push(`${name}\t${r.status === 0 ? "SURVIVED" : "KILLED"}\t${tests}\t${fails.join(" | ").slice(0, 400)}`);
  } finally { writeFileSync(p, orig, "utf8"); }
  console.log(res.at(-1));
}
