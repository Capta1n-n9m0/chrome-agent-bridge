## The probe script (template; __SDK__ replaced per way)
// Stage 0 probe: resolves the SDK, runs its token lookup, prints no token.
import { existsSync } from "node:fs";
import path from "node:path";
import { connect, BridgeClient, ErrorCodes } from "__SDK__";
import { resolveToken, moduleDir, findWorkspaceRoot } from "__SDK_TOKEN__";

const scratch = process.env.STAGE0_SCRATCH ?? "";
const clean = (p: string | null): string => (p === null ? "null" : scratch ? p.split(scratch).join("<scratch>") : p);

console.log("connect is a function:", typeof connect === "function");
console.log("BridgeClient is a class:", typeof BridgeClient === "function");
console.log("ErrorCodes.NOT_CONNECTED:", ErrorCodes.NOT_CONNECTED);
console.log("token module dir:", clean(moduleDir()));
const root = findWorkspaceRoot(moduleDir());
console.log("workspace root found:", clean(root));
const t = resolveToken({ env: {} });
console.log("token equals the clone's dummy:", t === "dummy-stage0-not-a-token");
console.log("daemon/dist/index.js under that root:", root !== null && existsSync(path.join(root, "daemon", "dist", "index.js")));
