import difflib, re, sys
sys.stdout.reconfigure(encoding="utf-8")
path = sys.argv[1]
lines = open(path, encoding="utf-8").read().split("\n")
def row(prefix):
    hits = [l for l in lines if l.startswith("| " + prefix + " ")]
    assert len(hits) == 1, (prefix, len(hits))
    return hits[0]
def cells(l):
    # split on " | " boundaries outside backticks
    out, cur, tick = [], "", False
    s = l.strip()
    assert s.startswith("|") and s.endswith("|")
    s = s[1:-1]
    for ch in s:
        if ch == "`":
            tick = not tick
        if ch == "|" and not tick:
            out.append(cur.strip()); cur = ""
        else:
            cur += ch
    out.append(cur.strip())
    return out
def toks(s):
    return re.findall(r"\s+|[A-Za-z0-9_\-\.<>≤]+|[^\sA-Za-z0-9_]", s)
for orig, new in [("R20-POP-BG", "R20-POP-BG-3"), ("R20-POP-FG", "R20-POP-FG-3"), ("R20-POP-OTHER", "R20-POP-OTHER-3")]:
    a, b = cells(row(orig)), cells(row(new))
    print("=" * 100)
    print(f"{orig} ({len(a)} cells)  vs  {new} ({len(b)} cells)")
    for i, (x, y) in enumerate(zip(a, b), 1):
        ta, tb = toks(x), toks(y)
        sm = difflib.SequenceMatcher(None, ta, tb, autojunk=False)
        ops = [op for op in sm.get_opcodes() if op[0] != "equal"]
        print(f"--- cell {i}: {len(ops)} differing span(s)")
        for k, (tag, i1, i2, j1, j2) in enumerate(ops, 1):
            print(f"  [{k}] {tag}")
            print(f"      OLD: {''.join(ta[i1:i2])!r}")
            print(f"      NEW: {''.join(tb[j1:j2])!r}")
