// Scratch Chrome over --remote-debugging-pipe (no port). launch(profileDir, {headless, unsafeExt}) -> {pid, send, on, close}.
import { spawn } from "node:child_process";
import fs from "node:fs";
export const CHROME = "<chrome>";
export const sleep = (ms) => new Promise((r) => setTimeout(r, ms));
export function launch(profile, { headless = true, unsafeExt = true, log = () => {} } = {}) {
  if (!profile || !profile.includes("\\stage0\\m1\\")) throw new Error("profile must be under stage0\\m1");
  const args = [`--user-data-dir=${profile}`, "--remote-debugging-pipe"];
  if (unsafeExt) args.push("--enable-unsafe-extension-debugging");
  if (headless) args.push("--headless=new");
  args.push("--no-first-run", "--no-default-browser-check", "--disable-background-networking", "--disable-component-update",
    "--disable-sync", "--disable-features=Translate,OptimizationHints,MediaRouter", "--window-size=1200,850", "about:blank");
  const cp = spawn(CHROME, args, { stdio: ["ignore", "ignore", "ignore", "pipe", "pipe"] });
  log(`spawned chrome pid=${cp.pid} args=${args.filter((a) => !a.startsWith("--user-data-dir")).join(" ")}`);
  const out = cp.stdio[3], inp = cp.stdio[4];
  let buf = "", nextId = 1; const waiters = new Map(); const listeners = [];
  inp.on("data", (d) => {
    buf += d.toString();
    let i;
    while ((i = buf.indexOf("\0")) >= 0) {
      const msg = JSON.parse(buf.slice(0, i)); buf = buf.slice(i + 1);
      if (msg.id && waiters.has(msg.id)) { waiters.get(msg.id)(msg); waiters.delete(msg.id); }
      else if (msg.method) for (const l of listeners) l(msg);
    }
  });
  const send = (method, params = {}, sessionId) => new Promise((res) => { const id = nextId++; waiters.set(id, res); const m = { id, method, params }; if (sessionId) m.sessionId = sessionId; out.write(JSON.stringify(m) + "\0"); });
  const exited = new Promise((res) => cp.on("exit", (code) => { log(`chrome exited code=${code}`); res(code); }));
  const close = async () => { const c = await send("Browser.close"); log(`Browser.close -> ${JSON.stringify(c)}`); return Promise.race([exited, sleep(10000).then(() => "timeout")]); };
  return { pid: cp.pid, send, on: (f) => listeners.push(f), close, exited };
}
// Attach (flatten) to a target and return its sessionId.
export async function attach(c, targetId) { const r = await c.send("Target.attachToTarget", { targetId, flatten: true }); return r.result.sessionId; }
export async function evalIn(c, sessionId, expr, extra = {}) {
  const r = await c.send("Runtime.evaluate", { expression: expr, awaitPromise: true, returnByValue: true, ...extra }, sessionId);
  if (r.error) return { cdpError: r.error.message };
  if (r.result.exceptionDetails) return { exception: r.result.exceptionDetails.exception?.description ?? r.result.exceptionDetails.text };
  return r.result.result.value;
}
export async function swSession(c, extId, tries = 50) {
  for (let i = 0; i < tries; i++) {
    const { result: { targetInfos } } = await c.send("Target.getTargets");
    const t = targetInfos.find((x) => x.type === "service_worker" && x.url.startsWith(`chrome-extension://${extId}/`));
    if (t) return attach(c, t.targetId);
    await sleep(200);
  }
  throw new Error("no service worker for " + extId);
}
export const writeJson = (f, v) => fs.writeFileSync(f, JSON.stringify(v, null, 1));
