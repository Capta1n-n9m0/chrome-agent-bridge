// usage: node variant.mjs <wt> <fixsrc> <variant>; writes the three src files for the variant (from the fix copies)
import { readFileSync, writeFileSync } from "node:fs";
import { execSync } from "node:child_process";
const [wt, fix, v] = process.argv.slice(2);
const files = { d: "extension/src/debugger.ts", r: "extension/src/evaluate/result.ts", s: "extension/src/evaluate/serialize.ts" };
const base = { d: "debugger.ts", r: "result.ts", s: "serialize.ts" };
const src = {};
for (const k of Object.keys(files)) src[k] = readFileSync(`${fix}/${base[k]}`, "utf8").replace(/\r\n/g, "\n");
function rep(k, a, b) {
  if (!src[k].includes(a)) throw new Error(`variant ${v}: missing ${a}`);
  src[k] = src[k].replace(a, b);
}
const WRAP_FIX = `"  return (" + String(envelopeJsonFn) + ")((" + SERIALIZER_SRC + ").call(this, limits));",`;
const WRAP_OLD = `"  return JSON.stringify((" + SERIALIZER_SRC + ").call(this, limits));",`;
const MARK_FIX = `    ("__mark" in env) === markRequested &&\n    (!markRequested || MARK_STATUSES.has(env.__mark as string));`;
const MARK_OLD = `    (!("__mark" in env) || MARK_STATUSES.has(env.__mark as string));`;
if (v === "preslice") {
  for (const k of Object.keys(files)) src[k] = execSync(`git -C "${wt}" show 518d6dc:${files[k]}`, { encoding: "utf8" });
} else if (v === "builder") {
  rep("s", WRAP_FIX, WRAP_OLD);
  rep("r", MARK_FIX, MARK_OLD);
} else if (v === "mF1") {
  rep("s", WRAP_FIX, WRAP_OLD);
} else if (v === "mF2") {
  rep("r", MARK_FIX, `    (!markRequested || "__mark" in env) &&\n` + MARK_OLD);
} else if (v === "mF3") {
  rep("r", MARK_FIX, `    (markRequested || !("__mark" in env)) &&\n` + MARK_OLD);
} else if (v === "mK21") {
  rep("d", `import { SERIALIZER_JSON_SRC, type SerializeLimits }`, `import { SERIALIZER_SRC, type SerializeLimits }`);
  rep("d", `functionDeclaration: SERIALIZER_JSON_SRC,`, `functionDeclaration: SERIALIZER_SRC,`);
  rep("d", `return parseEnvelopeText(res.result?.value, typeof lim.mark === "string" && lim.mark !== "");`, `return res.result?.value as EvalEnvelope;`);
} else if (v === "mWsort") {
  rep("s", "const keys = Object.keys(x);", "const keys = Object.keys(x).sort();");
} else if (v === "mWundef") {
  rep("s", "(s === undefined ? \"null\" : s)", "(s === undefined ? \"\" : s)");
} else if (v !== "fix") throw new Error("unknown variant");
for (const k of Object.keys(files)) writeFileSync(`${wt}/${files[k]}`, src[k]);
console.log("variant", v, "written");
