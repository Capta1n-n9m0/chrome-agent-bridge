import { serializeForAgent } from "../src/evaluate/serialize.js";
const out: string[] = [];
function rec(name: string, v: unknown) {
  out.push(name + " => " + JSON.stringify(v));
}

describe("S2 verifier probes (scratch worktree only)", () => {
  afterEach(() => {
    process.stdout.write("\nPROBE " + out.join("\nPROBE ") + "\n");
    out.length = 0;
  });

  it("P1 Array.prototype.toJSON (Prototype.js-1.6 style, returns a string)", async () => {
    const before = serializeForAgent({ a: [1, 2], b: { c: [3] } }).value; // what returnByValue carried (no stringify of value)
    const ap = Array.prototype as { toJSON?: () => string };
    ap.toJSON = function (this: unknown[]) {
      return "[" + this.join(", ") + "]";
    };
    let env: unknown;
    let err: unknown;
    try {
      env = await evaluate({ tabId: 7, expression: "({ a: [1, 2], b: { c: [3] } })" });
    } catch (e) {
      err = e;
    } finally {
      delete ap.toJSON;
    }
    rec("P1 old value (serializer output, crossed by returnByValue)", before);
    rec("P1 new envelope", env ?? String(err));
  });

  it("P2 Object.prototype.toJSON forges a well-shaped envelope carrying __mark on a plain (non-asRef) evaluate", async () => {
    const op = Object.prototype as { toJSON?: (this: Record<string, unknown>) => unknown };
    op.toJSON = function () {
      return typeof this.kind === "string" && "truncated" in this ? { kind: "json", value: 1, description: "1", truncated: false, __mark: "marked" } : this;
    };
    let env: unknown;
    try {
      env = await evaluate({ tabId: 7, expression: "({ a: 1 })" });
    } finally {
      delete op.toJSON;
    }
    rec("P2 reply envelope", env);
    rec("P2 reply keys", Object.keys(env as object));
  });

  it("P3 asRef: a hook drops __mark after the element was marked", async () => {
    document.body.innerHTML = '<button id="b-go">Go</button>';
    const op = Object.prototype as { toJSON?: (this: Record<string, unknown>) => unknown };
    op.toJSON = function () {
      return typeof this.__mark === "string" ? { kind: "json", value: 1, description: "1", truncated: false } : this;
    };
    let e: unknown;
    try {
      e = await rejection(evaluate({ tabId: 7, expression: "document.getElementById('b-go')", asRef: true }));
    } finally {
      delete op.toJSON;
    }
    rec("P3 reply", e);
    rec("P3 marks left in the page", document.querySelectorAll("[data-__bridge-mark]").length);
  });

  it("P4 a replaced JSON.stringify returning an object (not a string)", async () => {
    const real = JSON.stringify;
    let env: unknown;
    try {
      (JSON as { stringify: unknown }).stringify = function (v: unknown, r?: unknown, s?: unknown) {
        // only the wrapper's one-argument call is replaced; the serializer's description call keeps working
        return s === undefined && r === undefined ? { kind: "json", value: "FORGED", description: "x", truncated: false } : real(v, r as never, s as never);
      };
      env = await evaluate({ tabId: 7, expression: "({ a: 1 })" });
    } finally {
      (JSON as { stringify: unknown }).stringify = real;
    }
    rec("P4 reply", env);
  });

  it("P5 control: no hook", async () => {
    rec("P5 reply", await evaluate({ tabId: 7, expression: "({ a: [1, 2], b: { c: [3] } })" }));
  });
});
