// Runs the M2 pin's assertion logic (copied from extension/test/s3a-dnr-wiring.test.ts) against sw.ts and two mutants
// held in memory. The main tree's sw.ts is only read.
import { readFileSync } from "node:fs";
const src = readFileSync(process.argv[2], "utf8");
const strip = (s) => s.replace(/\/\*[\s\S]*?\*\//g, "").replace(/(^|[^:])\/\/.*$/gm, "$1");
function pin(code) {
  const fails = [];
  // The listener that builds the event (sw.ts has an earlier onRemoved listener too), up to its closing line.
    const listener = (on, built) => {
      const at = code.indexOf(built);
      const start = at < 0 ? -1 : code.lastIndexOf(on, at);
      return start < 0 ? "" : code.slice(start, code.indexOf("\n});", start));
    };
    const removed = listener("chrome.tabs.onRemoved.addListener(", "tabRemovedParams(");
  const replaced = listener("chrome.tabs.onReplaced.addListener(", "tabReplacedParams(");
  for (const [name, body, built, event] of [
    ["onRemoved", removed, "tabRemovedParams(", "\"tabRemoved\""],
    ["onReplaced", replaced, "tabReplacedParams(", "\"tabReplaced\""],
  ]) {
    const gone = body.indexOf("openerJoin.gone(");
    if (!(gone > 0)) fails.push(`${name}: gone missing`);
    if (!(gone < body.indexOf(built))) fails.push(`${name}: gone after ${built}`);
    if (!(gone < body.indexOf(event))) fails.push(`${name}: gone after ${event}`);
    if (!(gone < body.indexOf("sendTabEvent("))) fails.push(`${name}: gone after sendTabEvent(`);
  }
  return fails;
}
const goneR = "  openerJoin.gone(tabId); // Step 13 S1: a held tabCreated goes out first\n";
const sendR = "  sendTabEvent((v) => eventFrame(\"tabRemoved\", redactTabRemoved(params, v.switches, v.ctx)));\n";
const goneP = "  openerJoin.gone(removedTabId); // Step 13 S1: a held tabCreated goes out first\n";
const sendP = "  sendTabEvent(() => frame);\n});\n\n// Refill";
for (const s of [goneR, sendR, goneP, sendP]) if (!src.includes(s)) throw new Error("anchor missing: " + s);
// V13 (the verifier's mutant): gone moved after the tabRemoved send.
const mutA = src.replace(goneR, "").replace(sendR, sendR + goneR);
// The same swap in onReplaced.
const mutB = src.replace(goneP, "").replace(sendP, "  sendTabEvent(() => frame);\n" + goneP + "});\n\n// Refill");
for (const [name, s] of [["original", src], ["mutant A (onRemoved: gone after send)", mutA], ["mutant B (onReplaced: gone after send)", mutB]]) {
  const f = pin(strip(s));
  console.log(`${name}: ${f.length === 0 ? "GREEN" : "RED " + JSON.stringify(f)}`);
}
