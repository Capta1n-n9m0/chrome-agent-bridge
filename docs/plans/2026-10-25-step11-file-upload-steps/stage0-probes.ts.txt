type Answer = { result?: unknown; error?: Record<string, unknown>; delayMs?: number };

class FakeExt {
  received: Array<{ method: string; params: Record<string, unknown>; at: number }> = [];
  answered: Array<{ method: string; at: number }> = [];
  override = new Map<string, (p: Record<string, unknown>) => Answer>();
  tabs = [{ id: 101, title: "Human", url: "http://localhost:8080/e2e-playground.html", active: true, windowId: 1, current: true }];
  private next = 200;
  ws: WebSocket | undefined;
  connect(port: number, session?: string): Promise<void> {
    return new Promise((resolve, reject) => {
      const ws = new WebSocket(`ws://127.0.0.1:${port}`);
      this.ws = ws;
      ws.on("message", (raw) => {
        const f = JSON.parse(raw.toString());
        if (f.type === "welcome") return resolve();
        if (typeof f.id !== "string") return;
        const params = f.params ?? {};
        this.received.push({ method: f.method, params, at: Date.now() });
        const a = this.answer(f.method, params);
        const send = () => {
          this.answered.push({ method: f.method, at: Date.now() });
          ws.send(JSON.stringify(a.error ? { id: f.id, error: a.error } : { id: f.id, result: a.result }));
        };
        if (a.delayMs) setTimeout(send, a.delayMs);
        else send();
      });
      const hello: Record<string, unknown> = { type: "hello", token: TOKEN, role: "extension", proto: 2 };
      if (session) hello.session = session;
      ws.on("open", () => ws.send(JSON.stringify(hello)));
      ws.on("error", reject);
    });
  }
  methods(): string[] {
    return this.received.map((r) => r.method);
  }
  private answer(method: string, p: Record<string, unknown>): Answer {
    const o = this.override.get(method);
    if (o) return o(p);
    switch (method) {
      case "listTabs":
        return { result: { tabs: this.tabs } };
      case "newTab": {
        const id = this.next++;
        this.tabs.push({ id, title: "New", url: "http://localhost:8080/x", active: false, windowId: 1, current: undefined as never });
        return { result: { id } };
      }
      case "snapshot":
        return { result: { url: "http://localhost:8080/x", title: "x", elements: [], truncated: false } };
      case "screenshot":
        return { result: { dataUrl: PNG } };
      case "networkRequests":
        return { result: { entries: [], total: 0, pending: 0, recordingSince: 0 } };
      default:
        return { result: { ok: true } };
    }
  }
}

const cleanup: Array<() => Promise<void> | void> = [];
afterEach(async () => {
  for (const c of cleanup.splice(0).reverse()) await c();
});

async function setup(opts: { handleDir?: string; session?: string } = {}) {
  const root = mkdtempSync(path.join(os.tmpdir(), "bridge-s0probe-"));
  const daemonPath = path.join(root, "daemon.js");
  writeFileSync(daemonPath, "// not run");
  const logs: string[] = [];
  const bridge = new Bridge();
  const handleNumbers = opts.handleDir ? createHandleMark({ dir: opts.handleDir }) : undefined;
  const host = new WsHost(bridge, { port: 0, token: TOKEN, build: BUILD, log: (m) => logs.push(m), handleNumbers });
  const port = await host.listen();
  const ext = new FakeExt();
  await ext.connect(port, opts.session);
  const clients: BridgeClient[] = [];
  cleanup.push(async () => {
    for (const c of clients) { try { await Promise.resolve(c.close()); } catch { /* closed */ } }
    ext.ws?.terminate();
    await host.shutdown(1001);
    rmSync(root, { recursive: true, force: true });
  });
  const client = async (name: string, failureReports = true) => {
    const c = await connect({
      name,
      port,
      failureReports,
      env: { BRIDGE_TOKEN: TOKEN, BRIDGE_DAEMON: daemonPath, BRIDGE_FAILURES_DIR: root },
      io: { hashDaemon: async () => BUILD, spawnDaemon: async () => { throw new Error("no spawn in probe"); } },
    });
    clients.push(c);
    return c;
  };
  return { host, port, ext, logs, client, root };
}

async function codeOf(p: Promise<unknown>): Promise<BridgeError> {
  try {
    await p;
  } catch (e) {
    return e as BridgeError;
  }
  throw new Error("expected a rejection");
}

const OWN_ID = "abcdefghijklmnopabcdefghijklmnop";
const ALLOWED = "http://localhost:8080/e2e-playground.html";
type St = { armed?: boolean; evaluate?: boolean; list?: string; unreadable?: boolean };
function deps(s: St, tabs: Record<number, { url?: string }> = { 7: { url: ALLOWED } }): CallSafetyDeps {
  return {
    readState: vi.fn(async (): Promise<SafetyView> => {
      if (s.unreadable) throw new Error("storage read failed");
      const blockList = s.list ? parseBlockList(s.list) : EMPTY_BLOCK_LIST;
      return { switches: { armed: s.armed ?? true, evaluate: s.evaluate ?? true }, ctx: { extensionId: OWN_ID, blockList }, blockedCount: blockList.blockedCount };
    }),
    getTab: vi.fn(async (id: number) => {
      const t = tabs[id];
      if (!t) throw new Error(`No tab with id: ${id}.`);
      return t;
    }),
    replacedBy: () => undefined,
    lastUrl: () => undefined,
  };
}
