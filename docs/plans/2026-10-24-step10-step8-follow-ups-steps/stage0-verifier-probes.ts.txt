// ===== extension/test/zz-v-gate.test.ts (verifier's own; scratch only) =====
// Stage 0 VERIFIER throwaway probe (scratch worktree only; never committed). Independent of the auditor's
// probes: own harness, own cases. Results recorded in a Map (a plain object silently drops "__proto__").
import { describe, it, expect } from "vitest";
import { GatedRouter } from "../src/router.js";
import { CallSafety, type CallSafetyDeps } from "../src/safety-call.js";
import { METHOD_CLASSES } from "../src/safety.js";
import { parseBlockList, EMPTY_BLOCK_LIST } from "../src/safety-match.js";

const URL7 = "http://localhost:8080/e2e-playground.html";
function mkDeps(o: { armed?: boolean; js?: boolean; list?: string; fail?: boolean }, reads: { n: number }): CallSafetyDeps {
  return {
    readState: async () => {
      reads.n++;
      if (o.fail) throw new Error("read failed");
      const blockList = o.list ? parseBlockList(o.list) : EMPTY_BLOCK_LIST;
      return { switches: { armed: o.armed ?? true, evaluate: o.js ?? true }, ctx: { extensionId: "a".repeat(32), blockList }, blockedCount: blockList.blockedCount };
    },
    getTab: async (id: number) => {
      if (id === 7) return { url: URL7 };
      throw new Error(`No tab with id: ${id}.`);
    },
    replacedBy: () => undefined,
    lastUrl: (id: number) => (id === 7 ? URL7 : undefined),
  } as CallSafetyDeps;
}

describe("verifier probe — Step 9 extension contracts", () => {
  it("V-on: GatedRouter.on throws for prototype names and near-misses (recorded in a Map)", () => {
    const r = new GatedRouter(() => { throw new Error("gate factory must not be called by on()"); });
    const out = new Map<string, string>();
    for (const n of ["__proto__", "hasOwnProperty", "valueOf", "isPrototypeOf", "propertyIsEnumerable", "toLocaleString", "Evaluate", "evaluate ", "evaluateToFile", "selectTab"]) {
      try { r.on(n, async () => ({})); out.set(n, "REGISTERED"); } catch (e) { out.set(n, (e as Error).message); }
    }
    console.log(`VPROBE V-on ${JSON.stringify([...out])}`);
    expect([...out.values()].every((v) => v.startsWith("safety gate: method"))).toBe(true);
    expect(out.size).toBe(10);
  });

  it("V-unknown: unknown method → UNKNOWN_METHOD in every state; the gate factory is never invoked; positive control refuses", async () => {
    const states = { armed: {}, disarmed: { armed: false }, jsOff: { js: false }, unreadable: { fail: true }, blocked: { list: "localhost" } };
    const rows: string[] = [];
    for (const [label, s] of Object.entries(states)) {
      const reads = { n: 0 };
      let gates = 0;
      const r = new GatedRouter((m, p) => { gates++; return new CallSafety(m, p, mkDeps(s, reads)); });
      for (const m of Object.keys(METHOD_CLASSES)) r.on(m, async () => ({ ok: true }));
      for (const [m, p] of [["nope", { tabId: 7 }], ["__proto__", {}], ["constructor", { tabId: 7 }]] as const) {
        const rep = JSON.parse((await r.handle(JSON.stringify({ id: "q", method: m, params: p })))!);
        rows.push(`${label}/${m} ${rep.error?.code} gates=${gates} reads=${reads.n}`);
      }
      const ctl = JSON.parse((await r.handle(JSON.stringify({ id: "c", method: "evaluate", params: { tabId: 7, expression: "1" } })))!);
      rows.push(`${label}/control-evaluate ${ctl.error ? ctl.error.code + " " + JSON.stringify(ctl.error.data ?? null) : "ok"}`);
    }
    console.log(`VPROBE V-unknown\n  ${rows.join("\n  ")}`);
    expect(rows.filter((r) => !r.includes("control")).every((r) => / UNKNOWN_METHOD gates=0 reads=0$/.test(r))).toBe(true);
  });

  it("V-F4: CallSafety('evaluate') without tabId on a blocked tab — receipt, effect and post", async () => {
    const rows: string[] = [];
    for (const params of [{ id: "s1" }, { id: "s1", tabId: 7 }]) {
      const reads = { n: 0 };
      const c = new CallSafety("evaluate", params, mkDeps({ list: "localhost" }, reads));
      const rec = await c.receipt();
      let eff = "ok";
      try { await c.effect(); } catch (e) { eff = (e as { code?: string }).code ?? String(e); }
      const post = await c.post();
      rows.push(`${"tabId" in params ? "withTab" : "noTab"} receipt=${typeof rec === "string" ? rec : (rec as { code?: string }).code ?? JSON.stringify(rec)} effect=${eff} post=${post ? post.code : "null"}`);
    }
    console.log(`VPROBE V-F4\n  ${rows.join("\n  ")}`);
  });
});

// ===== mcp/test/zz-v-daemon.test.ts (verifier's own; scratch only) =====
// Stage 0 VERIFIER throwaway probe (scratch worktree only; never committed). Real WsHost on 127.0.0.1
// port 0, real SDK connect(), own fake proto:2 extension. Checks: (1) an unlisted method without `tab`
// is refused BAD_PARAMS and never forwarded; with `tab` it is forwarded with a numeric tabId and no `tab`;
// (2) while the last safetyState says disarmed, daemon.status carries no URL/title.
import { it, expect, afterEach } from "vitest";
import { mkdtempSync, rmSync, writeFileSync } from "node:fs";
import os from "node:os";
import path from "node:path";
import { WebSocket } from "ws";
import { connect } from "@bridge/sdk";
import { Bridge } from "../../daemon/src/bridge.js";
import { WsHost } from "../../daemon/src/wsHost.js";

const TOKEN = "verifier-probe-token";
const BUILD = "vbuild";
const done: Array<() => Promise<void> | void> = [];
afterEach(async () => { for (const f of done.splice(0).reverse()) await f(); });

it("V-daemon: untargeted unlisted method refused; disarmed status redacts", async () => {
  const dir = mkdtempSync(path.join(os.tmpdir(), "bridge-v0-"));
  const dpath = path.join(dir, "d.js");
  writeFileSync(dpath, "//");
  const host = new WsHost(new Bridge(), { port: 0, token: TOKEN, build: BUILD, log: () => {} });
  const port = await host.listen();
  const got: Array<{ method: string; params: Record<string, unknown> }> = [];
  const ext = new WebSocket(`ws://127.0.0.1:${port}`);
  await new Promise<void>((res, rej) => {
    ext.on("open", () => ext.send(JSON.stringify({ type: "hello", token: TOKEN, role: "extension", proto: 2, session: "S1" })));
    ext.on("error", rej);
    ext.on("message", (raw) => {
      const f = JSON.parse(raw.toString());
      if (f.type === "welcome") return res();
      if (typeof f.id !== "string") return;
      got.push({ method: f.method, params: f.params ?? {} });
      const result = f.method === "listTabs"
        ? { tabs: [{ id: 55, title: "Fixture Title", url: "http://localhost:8080/v.html", active: true, windowId: 1, current: true }] }
        : { ok: true };
      ext.send(JSON.stringify({ id: f.id, result }));
    });
  });
  const client = await connect({
    name: "v-probe", port,
    env: { BRIDGE_TOKEN: TOKEN, BRIDGE_DAEMON: dpath, BRIDGE_FAILURES_DIR: dir },
    io: { hashDaemon: async () => BUILD, spawnDaemon: async () => { throw new Error("no spawn"); } },
  } as never);
  done.push(async () => { try { await Promise.resolve(client.close()); } catch { /* */ } ext.terminate(); await host.shutdown(1001); rmSync(dir, { recursive: true, force: true }); });

  const rows: string[] = [];
  const before = got.length;
  try { await client.call("evaluateChunk", { id: "x" }); rows.push("untargeted: ok?!"); }
  catch (e) { const er = e as { code?: string; message?: string }; rows.push(`untargeted: ${er.code} "${er.message}" forwardedNew=${got.slice(before).map((g) => g.method).join(",") || "none"}`); }

  const t = await client.claim({ active: true } as never);
  const handle = (t as { handle: string }).handle;
  const b2 = got.length;
  await client.call("evaluateChunk", { tab: handle, id: "x" });
  const fw = got.slice(b2).find((g) => g.method === "evaluateChunk");
  rows.push(`targeted ${handle}: forwarded params=${JSON.stringify(fw?.params)}`);

  ext.send(JSON.stringify({ event: "safetyState", params: { armed: true, evaluate: true, blockedCount: 0 } }));
  await new Promise((r) => setTimeout(r, 150));
  const armed = JSON.stringify(await client.call("daemon.status", {}));
  ext.send(JSON.stringify({ event: "safetyState", params: { armed: false, evaluate: true, blockedCount: 0 } }));
  await new Promise((r) => setTimeout(r, 150));
  const dis = JSON.stringify(await client.call("daemon.status", {}));
  rows.push(`armed status: has '://'=${armed.includes("://")} hasTitle=${armed.includes("Fixture Title")}`);
  rows.push(`disarmed status: has '://'=${dis.includes("://")} hasTitle=${dis.includes("Fixture Title")} safety=${JSON.stringify(JSON.parse(dis).safety)}`);
  console.log(`VPROBE V-daemon\n  ${rows.join("\n  ")}`);
  expect(rows[0]).toMatch(/^untargeted: BAD_PARAMS .*tab is required.* forwardedNew=none$/);
  expect(fw?.params.tab).toBeUndefined();
  expect(typeof fw?.params.tabId).toBe("number");
  expect(dis.includes("://")).toBe(false);
});
