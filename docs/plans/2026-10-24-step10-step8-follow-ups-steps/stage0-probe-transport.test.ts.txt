// Step 10 Stage 0 THROWAWAY probe (scratch worktree only; never committed): phase doc 0.4's transport.
// The real daemon host (WsHost + Bridge + the tab layer) and the real SDK connect()/BridgeClient on
// 127.0.0.1 port 0, a fake proto:2 extension over real `ws`. The fake answers `evaluate` with the
// envelope the real extension builds for a string primitive (extension/src/evaluate/result.ts: `value`
// whole, `description` capped at 20 000 chars + the truncation line, `truncated: true`). Measures, per
// size, whether the value crosses extension -> daemon -> client whole, the time, and the process's peak
// RSS / heap (all three peers share this one process). Prints PROBE lines.
import { describe, it, expect, afterEach } from "vitest";
import { mkdtempSync, rmSync, writeFileSync } from "node:fs";
import os from "node:os";
import path from "node:path";
import { createHash } from "node:crypto";
import { WebSocket } from "ws";
import { connect, type BridgeClient, BridgeError } from "@bridge/sdk";
import { Bridge } from "../../daemon/src/bridge.js";
import { WsHost } from "../../daemon/src/wsHost.js";

const TOKEN = "stage0-transport-token-SENTINEL";
const BUILD = "stage0-build";
const MiB = 1024 * 1024;

class FakeExt {
  ws: WebSocket | undefined;
  closes: Array<{ code: number; reason: string }> = [];
  value = "";
  sentBytes = 0;
  connect(port: number): Promise<void> {
    return new Promise((resolve, reject) => {
      const ws = new WebSocket(`ws://127.0.0.1:${port}`);
      this.ws = ws;
      ws.on("close", (code, reason) => this.closes.push({ code, reason: reason.toString() }));
      ws.on("message", (raw) => {
        const f = JSON.parse(raw.toString());
        if (f.type === "welcome") return resolve();
        if (typeof f.id !== "string") return;
        let result: unknown = { ok: true };
        if (f.method === "listTabs") result = { tabs: [{ id: 101, title: "Human", url: "http://localhost:8080/e2e-playground.html", active: true, windowId: 1, current: true }] };
        if (f.method === "newTab") result = { id: 200 };
        if (f.method === "evaluate") {
          const v = this.value;
          const max = 20_000;
          result = v.length > max
            ? { kind: "json", value: v, description: v.slice(0, max) + `\n… [truncated: ${v.length - max} more chars]`, truncated: true }
            : { kind: "json", value: v, description: v, truncated: false };
        }
        const frame = JSON.stringify({ id: f.id, result });
        this.sentBytes = Buffer.byteLength(frame, "utf8");
        ws.send(frame);
      });
      ws.on("open", () => ws.send(JSON.stringify({ type: "hello", token: TOKEN, role: "extension", proto: 2 })));
      ws.on("error", () => {});
      setTimeout(() => reject(new Error("no welcome")), 5000);
    });
  }
}

const cleanup: Array<() => Promise<void> | void> = [];
afterEach(async () => {
  for (const c of cleanup.splice(0).reverse()) await c();
});

async function setup() {
  const root = mkdtempSync(path.join(os.tmpdir(), "bridge-s0transport-"));
  const daemonPath = path.join(root, "daemon.js");
  writeFileSync(daemonPath, "// not run");
  const logs: string[] = [];
  const host = new WsHost(new Bridge(), { port: 0, token: TOKEN, build: BUILD, log: (m) => logs.push(m) });
  const port = await host.listen();
  const ext = new FakeExt();
  await ext.connect(port);
  const clients: BridgeClient[] = [];
  cleanup.push(async () => {
    for (const c of clients) { try { await Promise.resolve(c.close()); } catch { /* closed */ } }
    ext.ws?.terminate();
    await host.shutdown(1001);
    rmSync(root, { recursive: true, force: true });
  });
  const c = await connect({
    name: "s0-transport",
    port,
    failureReports: false,
    env: { BRIDGE_TOKEN: TOKEN, BRIDGE_DAEMON: daemonPath, BRIDGE_FAILURES_DIR: root },
    io: { hashDaemon: async () => BUILD, spawnDaemon: async () => { throw new Error("no spawn in probe"); } },
  });
  clients.push(c);
  return { host, port, ext, c, logs };
}

/** One measurement: a value of `chars` copies of `ch`; returns a PROBE line. */
async function measure(label: string, chars: number, ch: string): Promise<{ whole: boolean; code?: string }> {
  const { ext, c, logs } = await setup();
  const tab = await c.newTab();
  ext.value = ch.repeat(chars);
  const want = createHash("sha256").update(ext.value, "utf8").digest("hex");
  global.gc?.();
  const base = process.memoryUsage();
  let peakRss = base.rss;
  let peakHeap = base.heapUsed;
  const sampler = setInterval(() => {
    const m = process.memoryUsage();
    peakRss = Math.max(peakRss, m.rss);
    peakHeap = Math.max(peakHeap, m.heapUsed);
  }, 20);
  const t0 = Date.now();
  let whole = false;
  let code: string | undefined;
  let got = "";
  try {
    const env = (await c.call("evaluate", { tab: tab.handle, expression: "x", timeoutMs: 60_000 }, { timeoutMs: 120_000 })) as { value: string };
    got = typeof env.value === "string" ? createHash("sha256").update(env.value, "utf8").digest("hex") : "not-a-string";
    whole = got === want && env.value.length === chars;
  } catch (e) {
    code = e instanceof BridgeError ? `${e.code}: ${e.message.slice(0, 120)}` : String(e).slice(0, 160);
  }
  const ms = Date.now() - t0;
  clearInterval(sampler);
  await new Promise((r) => setTimeout(r, 200));
  console.log(
    `PROBE 0.4 ${label} chars=${chars} utf8FrameBytes=${ext.sentBytes} (${(ext.sentBytes / MiB).toFixed(2)} MiB) whole=${whole} ` +
      `err=${code ?? "-"} ms=${ms} peakRssDeltaMiB=${((peakRss - base.rss) / MiB).toFixed(0)} peakHeapDeltaMiB=${((peakHeap - base.heapUsed) / MiB).toFixed(0)} ` +
      `extensionSocketCloses=${JSON.stringify(ext.closes)} daemonLogTail=${JSON.stringify(logs.slice(-2))}`,
  );
  // After a failure: is the daemon still usable by a fresh extension? (the real one would reconnect)
  return { whole, code };
}

describe("Stage 0 probe — 0.4 transport: one evaluate result, extension -> daemon -> client (port 0)", () => {
  it("ASCII strings of 1, 16, 64, 99 MiB characters", async () => {
    for (const mib of [1, 16, 64, 99]) await measure(`ascii-${mib}MiB`, mib * MiB, "a");
  }, 600_000);
  it("ASCII strings of 101 and 128 MiB characters (past ws's default maxPayload of 100 MiB)", async () => {
    for (const mib of [101, 128]) await measure(`ascii-${mib}MiB`, mib * MiB, "a");
  }, 600_000);
  it("non-ASCII: 40 Mi and 64 Mi characters of U+00E9 (2 UTF-8 bytes each) and 32 Mi of U+20AC (3 bytes)", async () => {
    await measure("e-acute-40Mi", 40 * MiB, "é");
    await measure("e-acute-64Mi", 64 * MiB, "é");
    await measure("euro-32Mi", 32 * MiB, "€");
  }, 600_000);
  it("control characters: 20 Mi characters of U+0001 (JSON escapes each as \\u0001, 6 bytes)", async () => {
    await measure("ctrl-20Mi", 20 * MiB, "\u0001");
  }, 600_000);
  it("the documented limit: ws maxPayload defaults (no option is passed anywhere in the repo)", async () => {
    const { WebSocketServer } = await import("ws");
    const s = new WebSocketServer({ noServer: true });
    console.log(`PROBE 0.4 ws.WebSocketServer default maxPayload=${(s.options as { maxPayload: number }).maxPayload}`);
    s.close();
    expect((s.options as { maxPayload: number }).maxPayload).toBe(100 * MiB);
  });
});
