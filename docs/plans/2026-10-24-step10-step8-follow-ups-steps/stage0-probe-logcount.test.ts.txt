import { it } from "vitest";
import { LOG_ALLOWLIST } from "./log-allowlist.js";
import { allLogSites, scannedSources } from "./log-scan.js";
it("count", () => {
  const by: Record<string, number> = {};
  for (const e of LOG_ALLOWLIST) { const r = e.file.split("/")[0]; by[r] = (by[r] ?? 0) + 1; }
  const found = allLogSites(scannedSources());
  const fby: Record<string, number> = {};
  for (const e of found) { const r = e.file.split("/")[0]; fby[r] = (fby[r] ?? 0) + 1; }
  console.log(`PROBE log-sites allowlist=${LOG_ALLOWLIST.length} ${JSON.stringify(by)} found=${found.length} ${JSON.stringify(fby)}`);
});
