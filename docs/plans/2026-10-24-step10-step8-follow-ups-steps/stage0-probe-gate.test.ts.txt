// Step 10 Stage 0 THROWAWAY probe (scratch worktree only; never committed): Step 9's extension-side
// contracts executed by importing the real modules — the GatedRouter, CallSafety, the method table, the
// sw.ts scan — and 0.4 (B)'s question: how a tab-less pull of class "evaluate" is gated. Prints PROBE lines.
import { describe, it, expect, vi } from "vitest";
import { readFileSync } from "node:fs";
import { GatedRouter } from "../src/router.js";
import { CallSafety, type CallSafetyDeps } from "../src/safety-call.js";
import { METHOD_CLASSES, methodClass, routerMethodNames } from "../src/safety.js";
import { parseBlockList, EMPTY_BLOCK_LIST } from "../src/safety-match.js";
import type { SafetyView } from "../src/safety-state.js";
import { DAEMON_ERROR_CODES, EXTENSION_ERROR_CODES, SDK_ERROR_CODES } from "@bridge/shared";

const OWN_ID = "abcdefghijklmnopabcdefghijklmnop";
const ALLOWED = "http://localhost:8080/e2e-playground.html";

type St = { armed?: boolean; evaluate?: boolean; list?: string; unreadable?: boolean };
function deps(s: St, tabs: Record<number, { url?: string }> = { 7: { url: ALLOWED } }): CallSafetyDeps {
  return {
    readState: vi.fn(async (): Promise<SafetyView> => {
      if (s.unreadable) throw new Error("storage read failed");
      const blockList = s.list ? parseBlockList(s.list) : EMPTY_BLOCK_LIST;
      return { switches: { armed: s.armed ?? true, evaluate: s.evaluate ?? true }, ctx: { extensionId: OWN_ID, blockList }, blockedCount: blockList.blockedCount };
    }),
    getTab: vi.fn(async (id: number) => {
      const t = tabs[id];
      if (!t) throw new Error(`No tab with id: ${id}.`);
      return t;
    }),
    replacedBy: () => undefined,
    lastUrl: () => undefined,
  };
}

const STATES: Record<string, St> = {
  armedJsOn: {},
  disarmed: { armed: false },
  jsOff: { evaluate: false },
  unreadable: { unreadable: true },
  blockedTab: { list: "localhost" },
};

describe("Step 10 Stage 0 probe — Step 9 extension-side contracts", () => {
  it("GatedRouter.on throws for every name without a safety class; accepts exactly the 23 classified ones; sw.ts registers exactly those 23", () => {
    const r = new GatedRouter((m, p) => new CallSafety(m, p, deps({})));
    const throwsFor: Record<string, string> = {};
    for (const name of ["evaluateToFile", "evaluateChunk", "evaluateEnd", "saveTo", "selectTab", "", "constructor", "__proto__", "toString"]) {
      try {
        r.on(name, async () => ({}));
        throwsFor[name] = "REGISTERED";
      } catch (e) {
        throwsFor[name] = (e as Error).message;
      }
    }
    for (const m of Object.keys(METHOD_CLASSES)) r.on(m, async () => ({}));
    const scan = routerMethodNames(readFileSync(new URL("../src/sw.ts", import.meta.url), "utf8"));
    console.log(`PROBE S9-on throws=${JSON.stringify(throwsFor)}`);
    console.log(`PROBE S9-on classified=${Object.keys(METHOD_CLASSES).length} swScan=${scan.names.length} nonLiteral=${JSON.stringify(scan.nonLiteral)} equal=${JSON.stringify([...scan.names].sort()) === JSON.stringify(Object.keys(METHOD_CLASSES).sort())}`);
    expect(Object.values(throwsFor).every((v) => v.startsWith("safety gate: method"))).toBe(true);
    expect(Object.keys(METHOD_CLASSES).length).toBe(23);
    expect([...scan.names].sort()).toEqual(Object.keys(METHOD_CLASSES).sort());
    expect(scan.nonLiteral).toEqual([]);
  });

  it("an unknown method is UNKNOWN_METHOD in every state, and the state is not even read", async () => {
    const out: Record<string, string> = {};
    for (const [label, s] of Object.entries(STATES)) {
      const d = deps(s);
      const r = new GatedRouter((m, p) => new CallSafety(m, p, d));
      for (const m of Object.keys(METHOD_CLASSES)) r.on(m, async () => ({ ok: true }));
      const reply = JSON.parse((await r.handle(JSON.stringify({ id: "1", method: "evaluateToFile", params: { tabId: 7 } })))!);
      out[label] = `${reply.error?.code} reads=${(d.readState as ReturnType<typeof vi.fn>).mock.calls.length}`;
    }
    console.log(`PROBE S9-unknown ${JSON.stringify(out)}`);
    for (const v of Object.values(out)) expect(v).toBe("UNKNOWN_METHOD reads=0");
  });

  it("DISARMED {scope}: disarmed → all; JavaScript off → evaluate for evaluate and a javascript:/data: navigate; unreadable → all; handlers not called", async () => {
    const rows: string[] = [];
    const cases: Array<[string, St, string, Record<string, unknown>]> = [
      ["evaluate/disarmed", { armed: false }, "evaluate", { tabId: 7, expression: "1" }],
      ["evaluate/jsOff", { evaluate: false }, "evaluate", { tabId: 7, expression: "1" }],
      ["evaluateAsRef/jsOff", { evaluate: false }, "evaluate", { tabId: 7, expression: "1", asRef: true }],
      ["navigate-javascript/jsOff", { evaluate: false }, "navigate", { tabId: 7, url: "javascript:void 0" }],
      ["newTab-data/jsOff", { evaluate: false }, "newTab", { url: "data:text/html,x" }],
      ["click/jsOff", { evaluate: false }, "click", { tabId: 7, ref: "e1" }],
      ["click/disarmed", { armed: false }, "click", { tabId: 7, ref: "e1" }],
      ["evaluate/unreadable", { unreadable: true }, "evaluate", { tabId: 7, expression: "1" }],
    ];
    for (const [label, s, method, params] of cases) {
      const calls: string[] = [];
      const r = new GatedRouter((m, p) => new CallSafety(m, p, deps(s)));
      for (const m of Object.keys(METHOD_CLASSES)) r.on(m, async () => { calls.push(m); return { ok: true }; });
      const reply = JSON.parse((await r.handle(JSON.stringify({ id: "1", method, params })))!);
      rows.push(`${label} -> ${reply.error ? `${reply.error.code} ${JSON.stringify(reply.error.data ?? null)}` : "ok"} handlerCalls=${calls.length}`);
    }
    console.log(`PROBE S9-scope\n  ${rows.join("\n  ")}`);
    expect(rows).toEqual([
      'evaluate/disarmed -> DISARMED {"scope":"all"} handlerCalls=0',
      'evaluate/jsOff -> DISARMED {"scope":"evaluate"} handlerCalls=0',
      'evaluateAsRef/jsOff -> DISARMED {"scope":"evaluate"} handlerCalls=0',
      'navigate-javascript/jsOff -> DISARMED {"scope":"evaluate"} handlerCalls=0',
      'newTab-data/jsOff -> DISARMED {"scope":"evaluate"} handlerCalls=0',
      "click/jsOff -> ok handlerCalls=1",
      'click/disarmed -> DISARMED {"scope":"all"} handlerCalls=0',
      'evaluate/unreadable -> DISARMED {"scope":"all"} handlerCalls=0',
    ]);
  });

  it("25 codes; the wire list is 24", () => {
    const total = DAEMON_ERROR_CODES.length + EXTENSION_ERROR_CODES.length + SDK_ERROR_CODES.length;
    console.log(`PROBE S9-codes(ext) daemon=${DAEMON_ERROR_CODES.length} extension=${EXTENSION_ERROR_CODES.length} sdk=${SDK_ERROR_CODES.length} total=${total}`);
    expect(total).toBe(25);
  });

  it("0.4 (B) question: a pull of class 'evaluate' with no tabId vs with a tabId, under each state (CallSafety receipt + post)", async () => {
    const rows: string[] = [];
    for (const [label, s] of Object.entries(STATES)) {
      for (const params of [{ id: "x" }, { id: "x", tabId: 7 }]) {
        const c = new CallSafety("evaluate", params, deps(s));
        const rec = await c.receipt();
        const post = await c.post();
        const show = (d: unknown) => (typeof d === "object" && d !== null && "code" in (d as object) ? `${(d as { code: string }).code} ${JSON.stringify((d as { data?: unknown }).data ?? null)}` : JSON.stringify(d));
        rows.push(`${label} ${"tabId" in params ? "withTab" : "noTab"}: receipt=${show(rec)} post=${show(post)}`);
      }
    }
    console.log(`PROBE 0.4B-class\n  ${rows.join("\n  ")}`);
    expect(methodClass("evaluate")).toBe("evaluate");
  });
});
