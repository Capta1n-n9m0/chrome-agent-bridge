// Step 9 Stage 0 THROWAWAY probe (deleted after the run; never committed). Re-executes Step 8's
// binding contracts over real `ws` on 127.0.0.1 port 0: the real daemon host, the real SDK connect(),
// a fake proto:2 extension. Prints PROBE lines that the evidence file quotes.
import { describe, it, expect, afterEach } from "vitest";
import { mkdtempSync, rmSync, writeFileSync } from "node:fs";
import os from "node:os";
import path from "node:path";
import { WebSocket } from "ws";
import { connect, type BridgeClient, BridgeError } from "@bridge/sdk";
import { Bridge } from "../../daemon/src/bridge.js";
import { WsHost } from "../../daemon/src/wsHost.js";
import { createHandleMark } from "../../daemon/src/handle-mark.js";
import { chooseMarker } from "../../extension/src/session-marker.js";
import { ATTACH_CODES } from "../../sdk/src/error.js";

const TOKEN = "stage0-probe-token-SENTINEL";
const BUILD = "stage0-build";
const PNG = "data:image/png;base64,iVBORw0KGgo=";

type Answer = { result?: unknown; error?: Record<string, unknown>; delayMs?: number };

class FakeExt {
  received: Array<{ method: string; params: Record<string, unknown>; at: number }> = [];
  answered: Array<{ method: string; at: number }> = [];
  override = new Map<string, (p: Record<string, unknown>) => Answer>();
  tabs = [{ id: 101, title: "Human", url: "http://localhost:8080/e2e-playground.html", active: true, windowId: 1, current: true }];
  private next = 200;
  ws: WebSocket | undefined;
  connect(port: number, session?: string): Promise<void> {
    return new Promise((resolve, reject) => {
      const ws = new WebSocket(`ws://127.0.0.1:${port}`);
      this.ws = ws;
      ws.on("message", (raw) => {
        const f = JSON.parse(raw.toString());
        if (f.type === "welcome") return resolve();
        if (typeof f.id !== "string") return;
        const params = f.params ?? {};
        this.received.push({ method: f.method, params, at: Date.now() });
        const a = this.answer(f.method, params);
        const send = () => {
          this.answered.push({ method: f.method, at: Date.now() });
          ws.send(JSON.stringify(a.error ? { id: f.id, error: a.error } : { id: f.id, result: a.result }));
        };
        if (a.delayMs) setTimeout(send, a.delayMs);
        else send();
      });
      const hello: Record<string, unknown> = { type: "hello", token: TOKEN, role: "extension", proto: 2 };
      if (session) hello.session = session;
      ws.on("open", () => ws.send(JSON.stringify(hello)));
      ws.on("error", reject);
    });
  }
  methods(): string[] {
    return this.received.map((r) => r.method);
  }
  private answer(method: string, p: Record<string, unknown>): Answer {
    const o = this.override.get(method);
    if (o) return o(p);
    switch (method) {
      case "listTabs":
        return { result: { tabs: this.tabs } };
      case "newTab": {
        const id = this.next++;
        this.tabs.push({ id, title: "New", url: "http://localhost:8080/x", active: false, windowId: 1, current: undefined as never });
        return { result: { id } };
      }
      case "snapshot":
        return { result: { url: "http://localhost:8080/x", title: "x", elements: [], truncated: false } };
      case "screenshot":
        return { result: { dataUrl: PNG } };
      case "networkRequests":
        return { result: { entries: [], total: 0, pending: 0, recordingSince: 0 } };
      default:
        return { result: { ok: true } };
    }
  }
}

const cleanup: Array<() => Promise<void> | void> = [];
afterEach(async () => {
  for (const c of cleanup.splice(0).reverse()) await c();
});

async function setup(opts: { handleDir?: string; session?: string } = {}) {
  const root = mkdtempSync(path.join(os.tmpdir(), "bridge-s0probe-"));
  const daemonPath = path.join(root, "daemon.js");
  writeFileSync(daemonPath, "// not run");
  const logs: string[] = [];
  const bridge = new Bridge();
  const handleNumbers = opts.handleDir ? createHandleMark({ dir: opts.handleDir }) : undefined;
  const host = new WsHost(bridge, { port: 0, token: TOKEN, build: BUILD, log: (m) => logs.push(m), handleNumbers });
  const port = await host.listen();
  const ext = new FakeExt();
  await ext.connect(port, opts.session);
  const clients: BridgeClient[] = [];
  cleanup.push(async () => {
    for (const c of clients) { try { await Promise.resolve(c.close()); } catch { /* closed */ } }
    ext.ws?.terminate();
    await host.shutdown(1001);
    rmSync(root, { recursive: true, force: true });
  });
  const client = async (name: string, failureReports = true) => {
    const c = await connect({
      name,
      port,
      failureReports,
      env: { BRIDGE_TOKEN: TOKEN, BRIDGE_DAEMON: daemonPath, BRIDGE_FAILURES_DIR: root },
      io: { hashDaemon: async () => BUILD, spawnDaemon: async () => { throw new Error("no spawn in probe"); } },
    });
    clients.push(c);
    return c;
  };
  return { host, port, ext, logs, client, root };
}

async function codeOf(p: Promise<unknown>): Promise<BridgeError> {
  try {
    await p;
  } catch (e) {
    return e as BridgeError;
  }
  throw new Error("expected a rejection");
}

describe("Stage 0 probe — Step 8 contracts (port 0, fake proto:2 extension)", () => {
  it("C1 attach set: exactly 7 codes attach (snapshot+screenshot+networkRequests); others attach nothing", async () => {
    expect([...ATTACH_CODES].sort()).toEqual(
      ["LOCATOR_AMBIGUOUS", "LOCATOR_NOT_FOUND", "NAV_FAILED", "OFFSCREEN", "PAGE_ERROR", "REF_STALE", "TIMEOUT"],
    );
    const { ext, client } = await setup();
    const c = await client("probe-attach");
    const tab = await c.newTab();
    const codes = [...ATTACH_CODES, "RESTRICTED_URL", "INTERNAL", "DISARMED", "DOMAIN_BLOCKED", "DEBUGGER_ATTACH"];
    for (const code of codes) {
      ext.override.set("click", () => ({ error: { code, message: `fake ${code}` } }));
      const before = ext.received.length;
      const err = await codeOf(tab.click("e1"));
      const after = ext.methods().slice(before + 1); // the calls after the click itself
      const attached = ["snapshot", "screenshot", "networkRequests"].every((m) => after.includes(m));
      console.log(`PROBE C1 code=${code} thrown=${err.code} attachCalls=${JSON.stringify(after.sort())} attached=${attached}`);
      expect(err.code).toBe(code);
      if ((ATTACH_CODES as readonly string[]).includes(code)) expect(attached).toBe(true);
      else expect(after).toEqual([]);
    }
  });

  it("C2 code required: a codeless proto:2 error is delivered as INTERNAL and logged once per connection", async () => {
    const { ext, client, logs } = await setup();
    const c = await client("probe-code");
    const tab = await c.newTab();
    ext.override.set("click", () => ({ error: { message: "no code here" } }));
    const e1 = await codeOf(c.call("click", { tab: tab.handle, ref: "e1" }));
    const e2 = await codeOf(c.call("click", { tab: tab.handle, ref: "e1" }));
    const lines = logs.filter((l) => l.startsWith("extension error without code"));
    console.log(`PROBE C2 e1=${e1.code}/${e1.message} e2=${e2.code} logLines=${JSON.stringify(lines)}`);
    expect(e1.code).toBe("INTERNAL");
    expect(e1.message).toBe("no code here");
    expect(lines).toEqual(["extension error without code: click"]);
  });

  it("C3 session marker: pure branches; same marker keeps handles, a changed marker retires them", async () => {
    const fresh = () => "FRESH";
    const branches = {
      workerRestart: chooseMarker({ session: "S", local: "L", startup: false }, fresh),
      chromeStart: chooseMarker({ local: "L", startup: true, installed: "update" }, fresh),
      reload: chooseMarker({ local: "L", startup: false, installed: "update" }, fresh),
      reloadNoLocal: chooseMarker({ startup: false, installed: "install" }, fresh),
      chromeUpdate: chooseMarker({ local: "L", startup: false, installed: "chrome_update" }, fresh),
      noEvent: chooseMarker({ local: "L", startup: false }, fresh),
    };
    console.log(`PROBE C3 pure=${JSON.stringify(branches)}`);
    expect(branches.workerRestart).toEqual({ marker: "S", source: "worker-restart" });
    expect(branches.chromeStart).toEqual({ marker: "FRESH", source: "chrome-start" });
    expect(branches.reload).toEqual({ marker: "L", source: "reload" });
    expect(branches.reloadNoLocal).toEqual({ marker: "FRESH", source: "reload" });
    expect(branches.chromeUpdate).toEqual({ marker: "FRESH", source: "chrome-update" });
    expect(branches.noEvent).toEqual({ marker: "FRESH", source: "no-event" });

    const { ext, client, logs, port } = await setup({ session: "A" });
    const c = await client("probe-session");
    const gone: unknown[] = [];
    c.on("tabGone", (p) => gone.push(p));
    const tab = await c.newTab();
    // Same marker: a reload-like reconnect keeps the handle.
    ext.ws?.terminate();
    await ext.connect(port, "A");
    await new Promise((r) => setTimeout(r, 100));
    const ok = await c.call("snapshot", { tab: tab.handle });
    // Changed marker: Chrome restarted.
    ext.ws?.terminate();
    await ext.connect(port, "B");
    await new Promise((r) => setTimeout(r, 100));
    const err = await codeOf(c.call("snapshot", { tab: tab.handle }));
    const retireLines = logs.filter((l) => l.startsWith("extension session changed"));
    console.log(`PROBE C3 sameMarker=${ok ? "ok" : "?"} changed=${err.code} tabGone=${JSON.stringify(gone)} log=${JSON.stringify(retireLines)}`);
    expect(err.code).toBe("TAB_GONE");
    expect(gone).toEqual([{ tab: tab.handle, lastUrl: "http://localhost:8080/x", lastTitle: "New" }]);
    expect(retireLines).toEqual(["extension session changed: 2 handles retired"]); // the human tab (resync) + the newTab
  });

  it("C4 handle numbers are never reused across daemons (handle-mark), and a stale handle is TAB_UNKNOWN", async () => {
    const dir = mkdtempSync(path.join(os.tmpdir(), "bridge-s0mark-"));
    cleanup.push(() => rmSync(dir, { recursive: true, force: true }));
    const a = await setup({ handleDir: dir });
    const ca = await a.client("probe-mark-a");
    const ta = await ca.newTab();
    await Promise.resolve(ca.close());
    await a.host.shutdown(1001);
    const b = await setup({ handleDir: dir });
    const cb = await b.client("probe-mark-b");
    const tb = await cb.newTab();
    const stale = await codeOf(cb.call("snapshot", { tab: ta.handle }));
    const na = Number(ta.handle.slice(1));
    const nb = Number(tb.handle.slice(1));
    console.log(`PROBE C4 daemonA=${ta.handle} daemonB=${tb.handle} staleOnB=${stale.code}`);
    expect(nb).toBeGreaterThan(na);
    expect(stale.code).toBe("TAB_UNKNOWN");
  });

  it("C6 TAB_HELD: another client's tab fails fast with {owner, since}; nothing reaches the extension", async () => {
    const { ext, client } = await setup();
    const a = await client("probe-owner");
    const b = await client("probe-intruder");
    const ta = await a.newTab();
    const before = ext.received.length;
    const err = await codeOf(b.call("snapshot", { tab: ta.handle }));
    const reached = ext.received.slice(before).map((r) => r.method);
    console.log(`PROBE C6 code=${err.code} data=${JSON.stringify(err.data)} reachedExtension=${JSON.stringify(reached)}`);
    expect(err.code).toBe("TAB_HELD");
    expect((err.data as { owner: string }).owner).toBe("probe-owner");
    expect(typeof (err.data as { since: unknown }).since).toBe("number");
    expect(reached).toEqual([]);
  });

  it("C7 visibility lock: FIFO; a waiter past its deadline gets LOCK_TIMEOUT {holder}; a queued call is forwarded after the holder", async () => {
    const { ext, client } = await setup();
    const a = await client("probe-lock-a");
    const b = await client("probe-lock-b");
    const c = await client("probe-lock-c");
    const d = await client("probe-lock-d");
    const [ta, tb, tc, td] = [await a.newTab(), await b.newTab(), await c.newTab(), await d.newTab()];
    ext.override.set("screenshot", () => ({ result: { dataUrl: PNG }, delayMs: 300 }));
    const t0 = Date.now();
    const pa = a.call("screenshot", { tab: ta.handle });
    await new Promise((r) => setTimeout(r, 20));
    const pb = b.call("screenshot", { tab: tb.handle });
    await new Promise((r) => setTimeout(r, 20));
    const pc = c.call("screenshot", { tab: tc.handle });
    await new Promise((r) => setTimeout(r, 20));
    const pd = codeOf(d.call("screenshot", { tab: td.handle }, { timeoutMs: 150 }));
    await Promise.all([pa, pb, pc]);
    const dErr = await pd;
    const shots = ext.received.filter((r) => r.method === "screenshot");
    const order = shots.map((s) => `${s.params.tabId}@${s.at - t0}`);
    const answers = ext.answered.filter((r) => r.method === "screenshot").map((s) => s.at - t0);
    console.log(`PROBE C7 forwardOrder=${JSON.stringify(order)} answeredAt=${JSON.stringify(answers)} waiterD=${dErr.code} data=${JSON.stringify(dErr.data)}`);
    expect(shots.map((s) => s.params.tabId)).toEqual([200, 201, 202]);
    // Each later screenshot is forwarded only after the previous one was answered (serialised).
    expect(shots[1].at).toBeGreaterThanOrEqual(ext.answered.filter((r) => r.method === "screenshot")[0].at);
    expect(shots[2].at).toBeGreaterThanOrEqual(ext.answered.filter((r) => r.method === "screenshot")[1].at);
    expect(dErr.code).toBe("LOCK_TIMEOUT");
    expect((dErr.data as { holder: string }).holder).toBe("probe-lock-a");
    expect(shots.find((s) => s.params.tabId === 203)).toBeUndefined();
  });
});

import { McpServer } from "@modelcontextprotocol/sdk/server/mcp.js";
import { registerTools, type ToolBridge } from "../src/tools/registry.js";

describe("Stage 0 probe — Step 9 premises", () => {
  it("P1 redacted listTabs rows (no title/url): the daemon keeps them; daemon.status still serves the cached url; tabGone still carries it", async () => {
    const { ext, client } = await setup();
    const c = await client("probe-redact");
    const t = await c.newTab(); // the daemon caches url+title from the listTabs resync / newTab path
    await c.call("listTabs");
    // From now on the extension redacts (as 0.3 proposes while disarmed): rows carry no title/url.
    ext.override.set("listTabs", () => ({ result: { tabs: ext.tabs.map((r) => ({ id: r.id, active: r.active, windowId: r.windowId })) } }));
    const listed = (await c.call("listTabs")) as { tabs: Array<Record<string, unknown>> };
    const status = (await c.call("daemon.status")) as { tabs: Array<{ tab: string; url?: string }> };
    const sdkRows = await c.tabs();
    console.log(`PROBE P1 listTabsViaDaemon=${JSON.stringify(listed.tabs.map((r) => Object.keys(r).sort()))}`);
    console.log(`PROBE P1 daemonStatusUrls=${JSON.stringify(status.tabs.map((r) => [r.tab, r.url]))}`);
    console.log(`PROBE P1 sdkTabs=${JSON.stringify(sdkRows.map((r) => [r.handle, r.title, r.url]))}`);
    expect(listed.tabs.length).toBe(2);
    expect(status.tabs.find((r) => r.tab === t.handle)?.url).toBe("http://localhost:8080/x");
  });

  it("P2 the MCP renders a redacted row as 'undefined — undefined' today (S3 must render redaction)", async () => {
    const bridge = {
      hasClient: true,
      reason: null as string | null,
      call: async (method: string) => {
        if (method === "listTabs") return { tabs: [{ id: 1, tab: "T1", active: true, windowId: 1, current: true }] };
        if (method === "tabs.claim") return { tab: "T1" };
        return { ok: true };
      },
      status: async () => ({ port: 0, build: "b", uptime: 1, extensionConnected: true, clients: [{ id: "c1", name: "probe", since: 0 }], tabs: [], lock: {} }),
    } satisfies ToolBridge;
    const server = new McpServer({ name: "t", version: "0" });
    registerTools(server, bridge);
    const reg = (server as unknown as { _registeredTools: Record<string, { handler: (a: object, e: object) => Promise<{ content: Array<{ text?: string }> }> }> })._registeredTools;
    const out = (await reg.browser_list_tabs.handler({}, {})).content[0].text;
    const st = (await reg.browser_status.handler({}, {})).content[0].text;
    console.log(`PROBE P2 browser_list_tabs=${JSON.stringify(out)}`);
    console.log(`PROBE P2 browser_status=${JSON.stringify(st)}`);
    expect(out).toContain("undefined — undefined");
  });

  it("P3 a new daemon→client event name reaches an SDK client's generic listener and breaks nothing; a new extension→daemon event is ignored", async () => {
    const { ext, client, host } = await setup();
    const c = await client("probe-event");
    const seen: string[] = [];
    c.on("event", (m: { event: string }) => seen.push(m.event));
    ext.ws?.send(JSON.stringify({ event: "safetyState", params: { armed: false, evaluate: true, blockedCount: 0 } }));
    await new Promise((r) => setTimeout(r, 50));
    const ok = await c.call("listTabs");
    console.log(`PROBE P3 clientSawEvents=${JSON.stringify(seen)} extensionEventsCounted=${host.stats().extensionEvents} listTabsAfter=${ok ? "ok" : "?"}`);
    expect(seen).toEqual([]); // today's daemon does not relay it: 0.6 needs daemon work (S3)
  });
});
